package bench

// The recorded side of a durable run, on the four tables migration 0004 and 0005 already
// give this package plus the artifact store. Nothing here adds a table or a column.
//
// The mapping is worth stating because it decides what a reader can ask:
//
// runs holds one row per run: the kind, the benchmark version it cited, the deployment
// id and generation it was driven against, its state, and the operation row that is its
// job record. detail_json holds the run's own account of its conditions: the aggregate,
// the ceilings it hit, the isolation scope it had, the traffic it observed, the
// capability claims it is willing to make, and the artifact keys of its documents.
//
// run_attempts holds one row per case repetition: the attempt number, its state, the
// deployment pair, the child's exit code when a child ran, the transport failure line,
// and summary_json, which is the attempt's full record: the observation as it was
// measured, the verdicts, the evidence keys, and every operational note. An attempt row
// is written *before* the request goes out, which is what makes a crash recoverable
// rather than merely survivable: a row that says "running" after the station restarted is
// a request whose outcome nobody can know, and that is a state the schema already has a
// word for.
//
// run_cases holds one row per generated case, keyed by the digest of its own canonical
// bytes, with the prompt held as an artifact key. Two runs over one seed therefore share
// their case rows, which is the point of comparing them at all.
//
// run_evidence holds one row per artifact a run recorded. That row is what keeps the
// bytes: the artifact store refuses to delete content a run_evidence row names, so
// recording is retention. Anything a run knows that has no column is a canonical JSON
// document in the artifact store, named by run_evidence, and this package says so in the
// code rather than inventing schema: the pins document, the hardware snapshot, the run
// record, the traffic observation, the comparison and the dossier.

import (
	"context"
	"crypto/sha256"
	"database/sql"

	"encoding/json"
	"errors"
	"fmt"
	"strings"
	"time"

	"runyard.local/runyard/internal/store"
)

// The run states, which are the runs.state and operations.state column's own words.
const (
	RunStateQueued      = "queued"
	RunStateRunning     = "running"
	RunStateSucceeded   = "succeeded"
	RunStateFailed      = "failed"
	RunStateCancelled   = "cancelled"
	RunStateInterrupted = "interrupted"
)

// The attempt states this package writes. The column allows more than these; the others
// belong to a lifecycle worker driving a process.
const (
	AttemptStateRunning   = "running"
	AttemptStateSucceeded = "succeeded"
	AttemptStateFailed    = "failed"
	AttemptStateCancelled = "cancelled"
	AttemptStateUnknown   = "unknown"
)

// The artifact kinds this package writes. They are the run_evidence kind column, so a
// reader can ask "give me every transport receipt of this run" as one query.
const (
	ArtifactKindPins        = "run-pins"
	ArtifactKindRecord      = "run-record"
	ArtifactKindReport      = "run-report"
	ArtifactKindAttempt     = "run-attempt-record"
	ArtifactKindRaw         = "run-attempt-raw"
	ArtifactKindEvaluator   = "run-evaluator-record"
	ArtifactKindTraffic     = "run-traffic-observation"
	ArtifactKindHardware    = "hardware-snapshot"
	ArtifactKindCase        = "run-case-prompt"
	ArtifactKindComparison  = "run-comparison"
	ArtifactKindDossier     = "run-dossier"
	ArtifactKindCaseDigests = "run-case-digests"
	ArtifactKindCaseKeys    = "run-case-keys"
)

// The operation kind a run job is recorded under.
const runOperationKind = "bench-run"

// RunAggregate is the run's four-way account. zero, unavailable, inapplicable and failed
// are four separate counters here, because they are four different things an operator
// acts on differently: a measured zero is an answer, an unavailable case is no answer at
// all, an inapplicable case was never askable, and a failed one was asked and the
// exchange broke.
type RunAggregate struct {
	CasesPlanned int `json:"casesPlanned"`
	CasesDriven  int `json:"casesDriven"`
	// Passes and Failures are the measured verdicts. Score is over these two only.
	Passes   int `json:"passes"`
	Failures int `json:"failures"`
	// MeasuredZero counts driven cases whose measured score is exactly zero. It is a
	// subset of Failures and it is spelled out because "everything scored zero" and
	// "nothing was measured" must not read the same.
	MeasuredZero int `json:"measuredZero"`
	// Inapplicable counts cases the target could not be judged on.
	Inapplicable int `json:"inapplicable"`
	// Blocked counts cases that did not get far enough to be judged.
	Blocked int `json:"blocked"`
	// Unavailable counts cases this run has no attempt for at all: it stopped, or a
	// ceiling cut them off. They are not zeros.
	Unavailable int `json:"unavailable"`
	// TransportFailures is a subset of Blocked: the cases whose exchange failed.
	TransportFailures int `json:"transportFailures"`
	// OperationalFailures counts the operational notes: ceilings, cancellations, refused
	// claims, evaluator failures. It is deliberately not mixed into the verdict counts.
	OperationalFailures int `json:"operationalFailures"`
	// RequestsMade, OutputBytes and ArtifactBytes are what the run consumed. They are
	// counted as they are spent, so a reader of a stopped run sees what it used.
	RequestsMade  int   `json:"requestsMade"`
	OutputBytes   int64 `json:"outputBytes"`
	ArtifactBytes int64 `json:"artifactBytes"`
	// Score is the fraction of measured verdicts that passed, and ScoreUnavailable says
	// why there is not one. Both are never set.
	Score            *float64 `json:"score,omitempty"`
	ScoreUnavailable string   `json:"scoreUnavailable,omitempty"`
	Coverage         float64  `json:"coverage"`
}

// CeilingReport is the run's bound set, recorded whether or not anything hit one: an
// operator comparing two runs needs to know the ceilings the numbers were taken under.
type CeilingReport struct {
	Ceilings Ceilings     `json:"ceilings"`
	Hits     []CeilingHit `json:"hits,omitempty"`
	// EffectiveConcurrency is the bound the run actually drove with, which is the smaller
	// of the benchmark version's concurrency and the run's ceiling.
	EffectiveConcurrency int  `json:"effectiveConcurrency"`
	PeakInFlight         int  `json:"peakInFlight"`
	Saturated            bool `json:"saturated,omitempty"`
}

// RunDetail is what runs.detail_json holds: the run's conditions and aggregate, small
// enough to read out of the row, and pointing at the documents that hold the rest.
type RunDetail struct {
	Aggregate     RunAggregate        `json:"aggregate"`
	AgenticPlan   *AgenticLevelPlan   `json:"agenticPlan,omitempty"`
	AgenticReport *AgenticLevelReport `json:"agenticReport,omitempty"`
	Ceilings      CeilingReport       `json:"ceilings"`
	Isolation     IsolationScope      `json:"isolation"`
	// Target is what the run drove, as it was given. It is recorded because a run's
	// declared role is part of what makes an inapplicable case mean something later, and
	// because the endpoint is what a reader of a two-week-old comparison checks against.
	Target       Target              `json:"target"`
	Traffic      *TrafficObservation `json:"traffic,omitempty"`
	Capabilities CapabilityStatement `json:"capabilities"`
	StoppedBy    string              `json:"stoppedBy,omitempty"`
	// WarnConcurrentRequests and TrafficWindowNanos are recorded so a rerun can ask the same
	// question about busy traffic with the same threshold and window, rather than deciding
	// them again and producing a run whose conditions nobody recorded.
	WarnConcurrentRequests int   `json:"warnConcurrentRequests,omitempty"`
	TrafficWindowNanos     int64 `json:"trafficWindowNanos,omitempty"`
	// Reruns is one entry per explicit rerun of this run, newest last. The attempts each
	// pass made stay where they are: a rerun adds attempt rows, it never edits an old one.
	Reruns   []RerunRecord `json:"reruns,omitempty"`
	Warnings []string      `json:"warnings,omitempty"`
	Notes    []string      `json:"notes,omitempty"`
	// Documents are the artifact keys of this run's own documents.
	Documents  map[string]string `json:"documents,omitempty"`
	PinsDigest string            `json:"pinsDigest,omitempty"`
	InputsHash string            `json:"inputsHash,omitempty"`
	Refusal    *RunRefusal       `json:"refusal,omitempty"`
}

// RerunRecord is one explicit rerun of a run: when, who asked, and which attempt numbers
// that pass added. The earlier attempts keep their own rows and their own records.
type RerunRecord struct {
	At           time.Time `json:"at"`
	Actor        string    `json:"actor"`
	FirstAttempt int       `json:"firstAttempt"`
	LastAttempt  int       `json:"lastAttempt"`
	Attempts     int       `json:"attempts"`
	State        string    `json:"state"`
}

// RunRefusal is the record of a run that never drove anything because the station refused
// it: an unapproved package, a registry with no such scorer. It is stored on the run, so a
// refused run reads as a refused run rather than as an empty one.
type RunRefusal struct {
	Stage  string `json:"stage"`
	Reason string `json:"reason"`
	Detail string `json:"detail,omitempty"`
}

// OperationNote is one operational fact about an attempt: that its exchange failed, that
// a ceiling stopped the run, that an evaluator claimed a pass over a failed exchange and
// was refused. Notes are never verdicts and never reach a score.
type OperationNote struct {
	Kind   string    `json:"kind"`
	Detail string    `json:"detail"`
	At     time.Time `json:"at"`
}

// The operational note kinds.
const (
	NoteTransport       = "transport_failure"
	NoteCeiling         = "ceiling"
	NoteCancelled       = "cancelled"
	NoteClaimRefused    = "evaluator_claim_refused"
	NoteEvaluatorFailed = "evaluator_failed"
	NoteRedaction       = "secret_redacted"
	NoteCapability      = "capability"
	NoteUnavailable     = "unavailable"
	NoteRecovery        = "recovered_after_restart"
	NoteRefusal         = "refused"
)

// CaseRecord is one attempt's whole record: what was sent, what came back, what was
// proved, what was judged, and what went wrong on the way. It is what run_attempts
// .summary_json holds, so an attempt row is self-describing without a join.
type CaseRecord struct {
	ExecutorDescriptor json.RawMessage `json:"executorDescriptor,omitempty"`
	RunID              string          `json:"runId"`
	AttemptNo          int             `json:"attemptNo"`
	CaseID             string          `json:"caseID"`
	Phase              string          `json:"phase,omitempty"`
	CaseKey            string          `json:"caseKey,omitempty"`
	Repetition         int             `json:"repetition"`
	Executor           string          `json:"executor,omitempty"`
	// Driven says a request was made. An attempt row with Driven false is an admission the
	// station never got to use, which is what a crash leaves behind, or a case the station
	// declined to attempt because the target cannot serve it.
	Driven bool `json:"driven"`
	// Transport is how the exchange ended, in one of four words, so a reader of a row does
	// not have to re-derive it from an observation.
	Transport string `json:"transport,omitempty"`
	// Cancelled says the run's own stop ended this attempt while it was open.
	Cancelled bool `json:"cancelled,omitempty"`
	// Recovered says the station restarted with this attempt still open, which is the only
	// way an attempt becomes outcome-unknown.
	Recovered bool `json:"recovered,omitempty"`
	// Observed is the transport receipt exactly as the executor reported it.
	Observed Observation `json:"observed"`
	// Response is the answer, with the raw payload left out: those bytes are their own
	// artifact, named in Evidence, so they are stored once and hashed once.
	Response Response `json:"response,omitempty"`
	// Failure is Observed.Failure(), kept as a field so a reader of the row sees it.
	Failure string `json:"failure,omitempty"`
	// Verdicts are the correctness verdicts, and only correctness. An operational
	// problem is a note and a blocked verdict, never a pass.
	Verdicts []Verdict `json:"verdicts,omitempty"`
	// Score is the fraction of this attempt's measured verdicts that passed, nil when
	// none were measured.
	Score *float64 `json:"score,omitempty"`
	// Notes are the operational facts.
	Notes []OperationNote `json:"notes,omitempty"`
	// Evidence names the bytes this attempt produced.
	Evidence CaseEvidence `json:"evidence"`
	// Evaluator is the child's own record, when a child was run.
	Evaluator  *EvaluatorAttemptRecord `json:"evaluator,omitempty"`
	StartedAt  time.Time               `json:"startedAt"`
	FinishedAt time.Time               `json:"finishedAt,omitempty"`
	// ScorerIDs names the scorers that judged this attempt, so the record says who judged
	// it without a reader having to open the version row.
	ScorerIDs []string `json:"scorers,omitempty"`
}

// CaseEvidence is the artifact keys of one attempt's bytes, with the redaction account
// beside them. A redacted artifact is not the same bytes that arrived, and the record says
// so instead of leaving a reader to find out by hashing.
type CaseEvidence struct {
	RawArtifactKey  string `json:"rawArtifactKey,omitempty"`
	RawBytes        int64  `json:"rawBytes,omitempty"`
	RecordArtifact  string `json:"recordArtifact,omitempty"`
	EvaluatorRecord string `json:"evaluatorRecordArtifact,omitempty"`
	Redactions      int    `json:"redactions,omitempty"`
	RedactedFrom    int64  `json:"redactedFromBytes,omitempty"`
}

// EvaluatorAttemptRecord is the child's own account, kept apart from the verdicts so that
// a child's exit status is never mistaken for a measurement.
type EvaluatorAttemptRecord struct {
	ID       string `json:"id"`
	ExitCode *int   `json:"exitCode,omitempty"`
	Signal   *int   `json:"signal,omitempty"`
	// ExitStatus is the host's own words for how the child ended. The two columns above
	// hold what could be read out of them; the text is kept because "signal: terminated"
	// is a fact this station did not observe a number for.
	ExitStatus string `json:"exitStatus,omitempty"`
	// Terminated says the station brought the process group down.
	Terminated bool             `json:"terminated,omitempty"`
	Pid        int              `json:"pid,omitempty"`
	Report     *EvaluatorReport `json:"report,omitempty"`
	Verdict    *Verdict         `json:"verdict,omitempty"`
	// ClaimAccepted says whether the station let the child's report stand as this case's
	// correctness verdict. A child that reported pass over an exchange that failed does not
	// get one, and this field says that it did not.
	ClaimAccepted bool          `json:"claimAccepted"`
	ClaimRefused  string        `json:"claimRefused,omitempty"`
	Negotiated    string        `json:"negotiated,omitempty"`
	CredentialID  string        `json:"credentialId,omitempty"`
	Artifacts     []ArtifactRef `json:"artifacts,omitempty"`
	Stderr        string        `json:"stderr,omitempty"`
	At            time.Time     `json:"at"`
}

// The four ways an attempt's exchange can end. They are recorded per attempt because the
// difference between them is the difference between a zero, nothing at all, and a broken
// wire, which is the distinction this whole checkpoint exists to keep.
const (
	// TransportAnswered: a payload came back, whatever it said.
	TransportAnswered = "answered"
	// TransportFailed: it did not: a transport error, a refusal, a timeout.
	TransportFailed = "failed"
	// TransportNotAttempted: the station declined to send anything, because the target
	// cannot serve what the case needs. The case is judged inapplicable; nothing failed.
	TransportNotAttempted = "not_attempted"
	// TransportUnknown: the station lost the thread, so nobody knows.
	TransportUnknown = "unknown"
)

// RowState is the attempt state the run_attempts row holds. A child's exit code, a signal
// and a retry count are recorded alongside it and decide nothing here: an attempt that
// exited 0 over an exchange that failed is a row that says failed, and that is the point of
// keeping both.
func (c CaseRecord) RowState() string {
	switch {
	case c.Recovered || c.Transport == TransportUnknown:
		return AttemptStateUnknown
	case c.Cancelled || c.Transport == TransportNotAttempted:
		// The run stopped this one, or declined to start it. Neither is an operational
		// failure and neither is an exchange that succeeded.
		return AttemptStateCancelled
	case c.Transport == TransportFailed || c.Failure != "":
		return AttemptStateFailed
	default:
		return AttemptStateSucceeded
	}
}

// caseKeyOf is the identity of a generated case: the digest of its own canonical bytes, so
// the same case generated by two runs is one row, and a case whose bytes differ is a
// different row rather than an edit of the first.
func caseKeyOf(c Case) (string, []byte, error) {
	canon, err := canonicalJSON(c)
	if err != nil {
		return "", nil, fmt.Errorf("bench: canonicalise case %s: %w", c.ID, err)
	}
	sum := sha256.Sum256(append([]byte(caseDigestDomain), canon...))
	return "case_" + hex.EncodeToString(sum[:])[:32], canon, nil
}

// caseDigestDomain names the case digest.
const caseDigestDomain = "runyard.bench.case.v1\n"

// caseSetDigest folds the case keys of a run into one digest, which is what the input hash
// covers: two runs with the same seed and a different generator revision drive different
// cases, and their scores are not two readings of one thing.
func caseSetDigest(keys []string) string {
	sum := sha256.Sum256([]byte(strings.Join(keys, "\n")))
	return hex.EncodeToString(sum[:])
}

// createRun writes the run's job record and its row together, so there is never a job with
// no run or a run no job is tracking. Both start queued: a run that claimed to be running
// before its first row committed would be a claim about a moment that did not happen.
func createRun(ctx context.Context, db *store.Store, actor, runID, kind, versionID string,
	deployment DeploymentPin, at time.Time) (string, error) {
	opID := runID + "-op"
	err := db.WriteAs(ctx, actor, "record a benchmark run", func(tx *store.Tx) error {
		if _, err := tx.Exec(`INSERT INTO operations
			(operation_id, kind, state, detail_json, created_at, updated_at) VALUES (?, ?, ?, ?, ?, ?)`,
			opID, runOperationKind, RunStateQueued,
			jsonOf(map[string]any{"runId": runID, "kind": kind}),
			stamp(at), stamp(at)); err != nil {
			return fmt.Errorf("bench: record the run's job row: %w", err)
		}
		if _, err := tx.Exec(`INSERT INTO runs
			(run_id, kind, benchmark_version_id, deployment_id, deployment_generation, state,
			 operation_id, detail_json, created_at)
			VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
			runID, kind, versionID, nullable(deployment.ID), nullableInt(deployment.Generation),
			RunStateQueued, opID, jsonOf(RunDetail{Isolation: IsolationUnknown}), stamp(at)); err != nil {
			return fmt.Errorf("bench: record the run: %w", err)
		}
		return nil
	})
	if err != nil {
		return "", err
	}
	return opID, nil
}

// setRunState moves the run and its job record together. The two rows say the same thing
// about the same run, so one of them moving alone would be a record with a contradiction
// in it.
func setRunState(ctx context.Context, db *store.Store, actor, runID, state, errorText string,
	detail any, at time.Time) error {
	payload := jsonOf(detail)
	return db.WriteAs(ctx, actor, "record the run's state", func(tx *store.Tx) error {
		var exists string
		err := tx.QueryRow(`SELECT run_id FROM runs WHERE run_id = ?`, runID).Scan(&exists)
		if errors.Is(err, sql.ErrNoRows) {
			return fmt.Errorf("bench: no run is recorded under %s", runID)
		}
		if err != nil {
			return fmt.Errorf("bench: read back the run %s: %w", runID, err)
		}
		var opID sql.NullString
		if err := tx.QueryRow(`SELECT operation_id FROM runs WHERE run_id = ?`, runID).Scan(&opID); err != nil {
			return fmt.Errorf("bench: read back the run %s job row: %w", runID, err)
		}
		res, err := tx.Exec(`UPDATE runs SET state = ?, error_text = ?, detail_json = ?,
			started_at = CASE WHEN started_at IS NULL AND ? THEN ? ELSE started_at END,
			finished_at = CASE WHEN ? THEN ? ELSE finished_at END
			WHERE run_id = ?`,
			state, nullable(errorText), payload,
			state == RunStateRunning, stamp(at),
			isTerminalRunState(state), stamp(at), runID)
		if err != nil {
			return fmt.Errorf("bench: record the run's state: %w", err)
		}
		if n, err := res.RowsAffected(); err == nil && n != 1 {
			return fmt.Errorf("bench: recording the run's state touched %d rows", n)
		}
		if !opID.Valid {
			return nil
		}
		if _, err := tx.Exec(`UPDATE operations SET state = ?, error_text = ?, detail_json = ?, updated_at = ?,
			finished_at = CASE WHEN ? THEN ? ELSE finished_at END WHERE operation_id = ?`,
			state, nullable(errorText), payload, stamp(at),
			isTerminalRunState(state), stamp(at), opID.String); err != nil {
			return fmt.Errorf("bench: record the run's job state: %w", err)
		}
		return nil
	})
}

// isTerminalRunState says whether a state ends the run.
func isTerminalRunState(state string) bool {
	switch state {
	case RunStateSucceeded, RunStateFailed, RunStateCancelled, RunStateInterrupted:
		return true
	}
	return false
}

// requestRunCancel sets the job record's cancellation flag, which is the durable half of a
// cancellation: a run driven by another process sees it too.
func requestRunCancel(ctx context.Context, db *store.Store, actor, runID string) error {
	return db.WriteAs(ctx, actor, "request a run be cancelled", func(tx *store.Tx) error {
		var opID sql.NullString
		err := tx.QueryRow(`SELECT operation_id FROM runs WHERE run_id = ?`, runID).Scan(&opID)
		if errors.Is(err, sql.ErrNoRows) {
			return fmt.Errorf("bench: no run is recorded under %s", runID)
		}
		if err != nil {
			return fmt.Errorf("bench: read back the run %s: %w", runID, err)
		}
		if !opID.Valid {
			return fmt.Errorf("bench: run %s has no job row to set a cancellation on", runID)
		}
		if _, err := tx.Exec(`UPDATE operations SET cancel_requested = 1, updated_at = ? WHERE operation_id = ?`,
			stamp(tx.At()), opID.String); err != nil {
			return fmt.Errorf("bench: request the run be cancelled: %w", err)
		}
		return nil
	})
}

// cancelRequested reads the durable cancellation flag.
func cancelRequested(ctx context.Context, db *store.Store, runID string) (bool, error) {
	var wanted int
	err := db.View(ctx, "read a run's cancellation", func(tx *store.Tx) error {
		err := tx.QueryRow(`SELECT o.cancel_requested FROM runs r JOIN operations o ON o.operation_id = r.operation_id
			WHERE r.run_id = ?`, runID).Scan(&wanted)
		if errors.Is(err, sql.ErrNoRows) {
			return nil
		}
		if err != nil {
			return fmt.Errorf("bench: read the cancellation of run %s: %w", runID, err)
		}
		return nil
	})
	return wanted == 1, err
}

// readRun reads one run row.
func readRun(ctx context.Context, db *store.Store, runID string) (RunRow, error) {
	var out RunRow
	err := db.View(ctx, "read a benchmark run", func(tx *store.Tx) error {
		row, err := readRunTx(tx, runID)
		if err != nil {
			return err
		}
		out = row
		return nil
	})
	return out, err
}

// readRunTx reads one run row from an open transaction, so a listing can read many rows
// without taking a reader connection per row.
func readRunTx(tx *store.Tx, runID string) (RunRow, error) {
	var out RunRow
	var (
		version, deployment, opID, errorText, detail sql.NullString
		generation                                   sql.NullInt64
		pinnedAt, pinnedBy                           sql.NullString
		startedAt, finishedAt                        sql.NullString
		createdAt                                    string
	)
	row := tx.QueryRow(`SELECT run_id, kind, benchmark_version_id, deployment_id, deployment_generation,
		state, operation_id, pinned_at, pinned_by, detail_json, error_text, created_at, started_at, finished_at
		FROM runs WHERE run_id = ?`, runID)
	if err := row.Scan(&out.RunID, &out.Kind, &version, &deployment, &generation, &out.State,
		&opID, &pinnedAt, &pinnedBy, &detail, &errorText, &createdAt, &startedAt, &finishedAt); err != nil {
		if errors.Is(err, sql.ErrNoRows) {
			return out, fmt.Errorf("bench: no run is recorded under %s", runID)
		}
		return out, fmt.Errorf("bench: read run %s: %w", runID, err)
	}
	out.BenchmarkVersionID = version.String
	out.DeploymentID = deployment.String
	out.DeploymentGeneration = int(generation.Int64)
	out.OperationID = opID.String
	out.ErrorText = errorText.String
	out.CreatedAt = parseStamp(createdAt)
	out.StartedAt = parseStamp(startedAt.String)
	out.FinishedAt = parseStamp(finishedAt.String)
	out.PinnedAt = parseStamp(pinnedAt.String)
	out.PinnedBy = pinnedBy.String
	if detail.String != "" {
		out.DetailRaw = json.RawMessage(detail.String)
		if err := json.Unmarshal([]byte(detail.String), &out.Detail); err != nil {
			return out, fmt.Errorf("bench: run %s holds a detail document this build cannot read: %w", runID, err)
		}
	}
	return out, nil
}

// listRuns lists recorded runs, newest first. Every row is read inside the one view, so a
// listing takes one reader connection rather than one per run.
func listRuns(ctx context.Context, db *store.Store, limit int) ([]RunRow, error) {
	if limit < 1 {
		limit = 50
	}
	var out []RunRow
	err := db.View(ctx, "list benchmark runs", func(tx *store.Tx) error {
		rows, err := tx.Query(`SELECT run_id FROM runs ORDER BY created_at DESC, run_id LIMIT ?`, limit)
		if err != nil {
			return fmt.Errorf("bench: list runs: %w", err)
		}
		defer rows.Close()
		var ids []string
		for rows.Next() {
			var id string
			if err := rows.Scan(&id); err != nil {
				return fmt.Errorf("bench: list runs: %w", err)
			}
			ids = append(ids, id)
		}
		if err := rows.Err(); err != nil {
			return fmt.Errorf("bench: list runs: %w", err)
		}
		rows.Close()
		for _, id := range ids {
			run, err := readRunTx(tx, id)
			if err != nil {
				return err
			}
			out = append(out, run)
		}
		return nil
	})
	return out, err
}

// nextAttemptNo is the attempt number the next admission would be given. A rerun records it
// so the run says which attempt numbers its newest pass added.
func nextAttemptNo(ctx context.Context, db *store.Store, runID string) (int, error) {
	no := 0
	err := db.View(ctx, "read the next attempt number of a run", func(tx *store.Tx) error {
		if err := tx.QueryRow(`SELECT COALESCE(MAX(attempt_no), 0) + 1 FROM run_attempts WHERE run_id = ?`,
			runID).Scan(&no); err != nil {
			return fmt.Errorf("bench: read the next attempt number of run %s: %w", runID, err)
		}
		return nil
	})
	return no, err
}

// admitAttempt writes an attempt row before its request goes out and returns the number it
// was given. The row is the admission: a station that dies after this point and before the
// response has a record that a request was made whose outcome nobody knows, which is the
// only honest thing it can know.
func admitAttempt(ctx context.Context, db *store.Store, actor, runID string,
	deployment DeploymentPin, rec CaseRecord, at time.Time) (int, error) {
	// The number is chosen inside the transaction, from the rows the transaction can
	// see, so it comes back as the write's result: a caller that gave up has no number,
	// because an attempt whose admission may still commit has not been given one.
	number, err := store.Write(db, ctx, actor, "admit a run attempt", func(tx *store.Tx) (int, error) {
		var no int
		if err := tx.QueryRow(`SELECT COALESCE(MAX(attempt_no), 0) + 1 FROM run_attempts WHERE run_id = ?`,
			runID).Scan(&no); err != nil {
			return 0, fmt.Errorf("bench: choose the attempt number of run %s: %w", runID, err)
		}
		rec.AttemptNo = no
		rec.StartedAt = at
		rec.RunID = runID
		body, err := json.Marshal(rec)
		if err != nil {
			return 0, fmt.Errorf("bench: encode the admitted attempt: %w", err)
		}
		_, err = tx.Exec(`INSERT INTO run_attempts
			(run_id, attempt_no, state, deployment_id, deployment_generation, summary_json, started_at)
			VALUES (?, ?, ?, ?, ?, ?, ?)`,
			runID, no, AttemptStateRunning, nullable(deployment.ID),
			nullableInt(deployment.Generation), string(body), stamp(at))
		if err != nil {
			return 0, fmt.Errorf("bench: admit attempt %d of run %s: %w", no, runID, err)
		}
		return no, nil
	})
	if err != nil {
		return 0, err
	}
	return number, nil
}

// finishAttempt records what an attempt turned into. The exit code and the signal come from
// a child process when one ran, and they are recorded next to the state rather than
// deciding it: a child that exited 0 over an exchange that failed is a row that says
// failed and 0, which is the whole point of keeping both.
func finishAttempt(ctx context.Context, db *store.Store, actor, runID string, no int,
	rec CaseRecord, at time.Time) error {
	body, err := json.Marshal(rec)
	if err != nil {
		return fmt.Errorf("bench: encode the attempt record: %w", err)
	}
	rec.FinishedAt = at
	body, err = json.Marshal(rec)
	if err != nil {
		return fmt.Errorf("bench: encode the attempt record: %w", err)
	}
	return db.WriteAs(ctx, actor, "record a run attempt", func(tx *store.Tx) error {
		res, err := tx.Exec(`UPDATE run_attempts SET state = ?, summary_json = ?, error_text = ?,
			exit_code = ?, terminated_by_signal = ?, finished_at = ?
			WHERE run_id = ? AND attempt_no = ?`,
			rec.RowState(), string(body), nullable(rec.Failure),
			nullableIntPtr(rec.Evaluator.exitCode()), nullableIntPtr(rec.Evaluator.signal()),
			stamp(at), runID, no)
		if err != nil {
			return fmt.Errorf("bench: record attempt %d of run %s: %w", no, runID, err)
		}
		if n, err := res.RowsAffected(); err == nil && n != 1 {
			return fmt.Errorf("bench: recording attempt %d of run %s touched %d rows", no, runID, n)
		}
		return nil
	})
}

// exitCode and signal reach into an absent evaluator record without every caller having
// to check for one.
func (e *EvaluatorAttemptRecord) exitCode() *int {
	if e == nil {
		return nil
	}
	return e.ExitCode
}

func (e *EvaluatorAttemptRecord) signal() *int {
	if e == nil {
		return nil
	}
	return e.Signal
}

// readAttempts reads a run's attempts in order, with each attempt's record decoded.
func readAttempts(ctx context.Context, db *store.Store, runID string) ([]AttemptRow, error) {
	var out []AttemptRow
	err := db.View(ctx, "read a run's attempts", func(tx *store.Tx) error {
		rows, err := tx.Query(`SELECT attempt_no, state, exit_code, terminated_by_signal, error_text,
			started_at, finished_at, summary_json FROM run_attempts WHERE run_id = ? ORDER BY attempt_no`, runID)
		if err != nil {
			return fmt.Errorf("bench: read the attempts of run %s: %w", runID, err)
		}
		defer rows.Close()
		for rows.Next() {
			var (
				row                        AttemptRow
				exit, sig                  sql.NullInt64
				errorText, started, finish sql.NullString
				summary                    sql.NullString
			)
			if err := rows.Scan(&row.AttemptNo, &row.State, &exit, &sig, &errorText,
				&started, &finish, &summary); err != nil {
				return fmt.Errorf("bench: read the attempts of run %s: %w", runID, err)
			}
			row.RunID = runID
			if exit.Valid {
				v := int(exit.Int64)
				row.ExitCode = &v
			}
			if sig.Valid {
				v := int(sig.Int64)
				row.Signal = &v
			}
			row.ErrorText = errorText.String
			row.StartedAt = parseStamp(started.String)
			row.FinishedAt = parseStamp(finish.String)
			if summary.String != "" {
				if err := json.Unmarshal([]byte(summary.String), &row.Record); err != nil {
					return fmt.Errorf("bench: attempt %d of run %s holds a record this build cannot read: %w",
						row.AttemptNo, runID, err)
				}
			}
			out = append(out, row)
		}
		return rows.Err()
	})
	return out, err
}

// recoverAttempts moves every attempt that was still in flight when the station stopped
// into the outcome-unknown state, and says so in the attempt's own record. It never
// re-drives anything: an inference request that the station cannot prove did or did not
// reach the endpoint is a request that may have been answered, billed, cached or counted,
// and repeating it on a guess is how a benchmark silently becomes a different run.
func recoverAttempts(ctx context.Context, db *store.Store, actor, runID string, at time.Time) (int, error) {
	// The count is the transaction's own, and it comes back with the transaction's
	// outcome, which is how a caller that waited still learns how far the recovery got
	// when it failed part way through.
	changed, err := store.Write(db, ctx, actor, "recover a run's in-flight attempts", func(tx *store.Tx) (int, error) {
		rows, err := tx.Query(`SELECT attempt_no, summary_json FROM run_attempts
			WHERE run_id = ? AND state IN (?, ?, ?, ?) ORDER BY attempt_no`,
			runID, AttemptStateRunning, "queued", "preparing", "starting")
		if err != nil {
			return 0, fmt.Errorf("bench: read the in-flight attempts of run %s: %w", runID, err)
		}
		defer rows.Close()
		type pending struct {
			no     int
			body   string
			record CaseRecord
		}
		var list []pending
		for rows.Next() {
			var p pending
			var body sql.NullString
			if err := rows.Scan(&p.no, &body); err != nil {
				return 0, fmt.Errorf("bench: read the in-flight attempts of run %s: %w", runID, err)
			}
			p.body = body.String
			if p.body != "" {
				if err := json.Unmarshal([]byte(p.body), &p.record); err != nil {
					return 0, fmt.Errorf("bench: attempt %d of run %s holds a record this build cannot read: %w",
						p.no, runID, err)
				}
			}
			list = append(list, p)
		}
		if err := rows.Err(); err != nil {
			return 0, fmt.Errorf("bench: read the in-flight attempts of run %s: %w", runID, err)
		}
		var changed int
		for _, p := range list {
			rec := p.record
			if rec.RunID == "" {
				rec.RunID = runID
			}
			if rec.AttemptNo == 0 {
				rec.AttemptNo = p.no
			}
			rec.FinishedAt = at
			// The record has to re-derive the state the row is being moved to, or a later reader
			// recomputing it from the record would get a different answer than the row holds.
			rec.Recovered = true
			rec.Transport = TransportUnknown
			rec.Driven = true
			if rec.Failure == "" {
				rec.Failure = "the station stopped while this request was open, so its outcome cannot be known"
			}
			rec.Notes = append(rec.Notes, OperationNote{
				Kind: NoteRecovery, At: at,
				Detail: "recovered after the station restarted as outcome-unknown: it is not re-run automatically, " +
					"a rerun is an explicit request and makes a new attempt",
			})
			body, err := json.Marshal(rec)
			if err != nil {
				return changed, fmt.Errorf("bench: encode the recovered attempt record: %w", err)
			}
			if _, err := tx.Exec(`UPDATE run_attempts SET state = ?, summary_json = ?, error_text = ?, finished_at = ?
				WHERE run_id = ? AND attempt_no = ?`,
				AttemptStateUnknown, string(body), nullable(rec.Failure), stamp(at), runID, p.no); err != nil {
				return changed, fmt.Errorf("bench: recover attempt %d of run %s: %w", p.no, runID, err)
			}
			changed++
		}
		return changed, nil
	})
	return changed, err
}

// recordCase writes one generated case: its canonical bytes as an artifact, and the row
// that names them. A case whose key is already held under different bytes is reported,
// because that is the station finding a generator that is not reproducible.
func recordCase(ctx context.Context, db *store.Store, actor string, at time.Time, versionID string,
	c Case, key string, promptKey string, params json.RawMessage) error {
	return db.WriteAs(ctx, actor, "record a run case", func(tx *store.Tx) error {
		var held sql.NullString
		err := tx.QueryRow(`SELECT prompt_artifact_key FROM run_cases WHERE case_key = ?`, key).Scan(&held)
		switch {
		case err == nil:
			if held.Valid && held.String != promptKey {
				return fmt.Errorf("bench: case %s is already recorded from %s, and this run generated %s: the generator is not reproducible",
					c.ID, held.String, promptKey)
			}
			return nil
		case !errors.Is(err, sql.ErrNoRows):
			return fmt.Errorf("bench: read back case %s: %w", c.ID, err)
		}
		_, err = tx.Exec(`INSERT INTO run_cases (case_key, name, benchmark_version_id, prompt_artifact_key,
			params_json, created_at) VALUES (?, ?, ?, ?, ?, ?)`,
			key, c.ID, nullable(versionID), nullable(promptKey), string(params), stamp(at))
		if err != nil {
			return fmt.Errorf("bench: record case %s: %w", c.ID, err)
		}
		return nil
	})
}

// readCases reads a run's cases by key. The keys come from the run's case digest document,
// so the order is the run's order rather than whatever the table returns.
func readCases(ctx context.Context, db *store.Store, keys []string) ([]CaseRow, error) {
	var out []CaseRow
	err := db.View(ctx, "read run cases", func(tx *store.Tx) error {
		for _, key := range keys {
			var (
				row                     CaseRow
				version, prompt, params sql.NullString
				createdAt               string
			)
			err := tx.QueryRow(`SELECT case_key, name, benchmark_version_id, prompt_artifact_key, params_json,
				created_at FROM run_cases WHERE case_key = ?`, key).
				Scan(&row.CaseKey, &row.Name, &version, &prompt, &params, &createdAt)
			if errors.Is(err, sql.ErrNoRows) {
				return fmt.Errorf("bench: no run case is recorded under %s", key)
			}
			if err != nil {
				return fmt.Errorf("bench: read run case %s: %w", key, err)
			}
			row.BenchmarkVersionID = version.String
			row.PromptArtifactKey = prompt.String
			row.Params = json.RawMessage(params.String)
			row.CreatedAt = parseStamp(createdAt)
			out = append(out, row)
		}
		return nil
	})
	return out, err
}

		if err != nil {

}

// readProgress counts a run's progress from its rows. Nothing is kept in memory, so a
// reader in another process sees the same numbers as the driver does.
func readProgress(ctx context.Context, db *store.Store, runID string) (RunProgress, error) {
	var out RunProgress
	err := db.View(ctx, "read a run's progress", func(tx *store.Tx) error {
		var (
			state             string
			detail, errorText sql.NullString
			created, started  sql.NullString
		)
		err := tx.QueryRow(`SELECT state, detail_json, error_text, created_at, started_at FROM runs WHERE run_id = ?`,
			runID).Scan(&state, &detail, &errorText, &created, &started)
		if errors.Is(err, sql.ErrNoRows) {
			return fmt.Errorf("bench: no run is recorded under %s", runID)
		}
		if err != nil {
			return fmt.Errorf("bench: read the progress of run %s: %w", runID, err)
		}
		out.RunID = runID
		out.State = state
		out.ErrorText = errorText.String
		out.StartedAt = parseStamp(started.String)
		out.UpdatedAt = parseStamp(created.String)
		var detailValue RunDetail
		if detail.String != "" {
			if err := json.Unmarshal([]byte(detail.String), &detailValue); err != nil {
				return fmt.Errorf("bench: run %s holds a detail document this build cannot read: %w", runID, err)
			}
		}
		out.CeilingsHit = nil
		for _, hit := range detailValue.Ceilings.Hits {
			out.CeilingsHit = append(out.CeilingsHit, hit.Name)
		}
		out.LimitConcurrent = detailValue.Ceilings.EffectiveConcurrency
		out.Warnings = detailValue.Warnings
		out.StoppedBy = detailValue.StoppedBy

		out.AttemptStates = map[string]int{}
		rows, err := tx.Query(`SELECT state, COUNT(*) FROM run_attempts WHERE run_id = ? GROUP BY state`, runID)
		if err != nil {
			return fmt.Errorf("bench: count the attempts of run %s: %w", runID, err)
		}
		defer rows.Close()
		for rows.Next() {
			var s string
			var n int
			if err := rows.Scan(&s, &n); err != nil {
				return fmt.Errorf("bench: count the attempts of run %s: %w", runID, err)
			}
			out.AttemptStates[s] = n
			out.Attempts += n
		}
		if err := rows.Err(); err != nil {
			return fmt.Errorf("bench: count the attempts of run %s: %w", runID, err)
		}
		out.CasesTotal = detailValue.Aggregate.CasesPlanned
		out.CasesDriven = detailValue.Aggregate.CasesDriven
		out.Passes = detailValue.Aggregate.Passes
		out.Failures = detailValue.Aggregate.Failures
		out.Inapplicable = detailValue.Aggregate.Inapplicable
		out.Blocked = detailValue.Aggregate.Blocked
		out.Unavailable = detailValue.Aggregate.Unavailable
		out.RequestsMade = detailValue.Aggregate.RequestsMade
		out.OutputBytes = detailValue.Aggregate.OutputBytes
		out.ArtifactBytes = detailValue.Aggregate.ArtifactBytes
		return nil
	})
	return out, err
}

// stamp renders a time the way this station writes them everywhere.
func stamp(t time.Time) string {
	if t.IsZero() {
		return ""
	}
	return t.UTC().Format(time.RFC3339Nano)
}

// parseStamp reads one back. An empty or unreadable stamp is the zero time, which is what
// a nullable timestamp column that holds NULL means.
func parseStamp(s string) time.Time {
	if s == "" {
		return time.Time{}
	}
	t, err := time.Parse(time.RFC3339Nano, s)
	if err != nil {
		return time.Time{}
	}
	return t.UTC()
}

// jsonOf encodes a document, or says so. A document that does not encode is a bug in this
// package, and a run row with a silently missing detail document would be worse than the
// bug.
func jsonOf(v any) string {
	raw, err := json.Marshal(v)
	if err != nil {
		panic("bench: encode a run document: " + err.Error())
	}
	return string(raw)
}

// nullableInt writes 0 as NULL, because these columns mean "not stated" rather than zero.
func nullableInt(v int) any {
	if v == 0 {
		return nil
	}
	return v
}

// nullableIntPtr passes a pointer through as a value or NULL.
func nullableIntPtr(v *int) any {
	if v == nil {
		return nil
	}
	return *v
}
