"""Deterministic external workloads used by the real-process suite."""
import argparse
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
import json
import os
from pathlib import Path
import time

parser = argparse.ArgumentParser()
parser.add_argument("mode", choices=("exit", "http"))
parser.add_argument("--counter")
parser.add_argument("--delay", type=float, default=1.5)
parser.add_argument("--port", type=int, default=0)
parser.add_argument("--metrics-file")
parser.add_argument("--metrics-content-type", default="text/plain")
parser.add_argument("--model-id", default="fixture-model")
parser.add_argument("--require-model", action="store_true")
parser.add_argument("--secret-env")
args = parser.parse_args()
if args.secret_env:
    print("diagnostic " + os.environ[args.secret_env], flush=True)
    print("bounded diagnostic is available", flush=True)
if args.counter:
    with Path(args.counter).open("a") as counter:
        counter.write("start\n")
if args.mode == "exit":
    print("fixture started", flush=True)
    time.sleep(args.delay)
    raise SystemExit(7)

class Handler(BaseHTTPRequestHandler):
    def do_GET(self):
        if self.path == "/metrics":
            value = Path(args.metrics_file).read_text() if args.metrics_file else "17"
            if value == "unavailable":
                self.send_error(503)
                return
            data = ("# TYPE runyard_generic_override gauge\nrunyard_generic_override 1\n"
                    "# TYPE queue_depth gauge\nqueue_depth " + value + "\n").encode()
        else:
            data = json.dumps({"object": "list", "data": [{"id": args.model_id, "object": "model"}]}).encode()
        self.send_response(200)
        self.send_header("Content-Type", args.metrics_content_type if self.path == "/metrics" else "application/json")
        self.send_header("Content-Length", str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def do_POST(self):
        request = json.loads(self.rfile.read(int(self.headers.get("Content-Length", "0"))))
        if args.require_model and request.get("model") != args.model_id:
            self.send_error(404, "unknown model")
            return
        reply = {"id": "fixture", "object": "chat.completion", "model": args.model_id,
                 "choices": [{"index": 0, "message": {"role": "assistant", "content": "fixture-answer"}, "finish_reason": "stop"}],
                 "usage": {"prompt_tokens": 10, "completion_tokens": 2, "total_tokens": 12}}
        self.send_response(200)
        self.send_header("Content-Type", "text/event-stream" if request.get("stream") else "application/json")
        self.end_headers()
        if request.get("stream"):
            reply["choices"] = [{"index": 0, "delta": {"role": "assistant", "content": "fixture-answer"}, "finish_reason": "stop"}]
            self.wfile.write(("data: " + json.dumps(reply) + "\n\ndata: [DONE]\n\n").encode())
        else:
            self.wfile.write(json.dumps(reply).encode())

ThreadingHTTPServer(("127.0.0.1", args.port), Handler).serve_forever()
