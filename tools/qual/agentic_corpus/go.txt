package bench

// The run's own bounds, enforced as it goes.
//
// A benchmark run asks a station to spend real resources on somebody's hypothesis, so
// each of the five bounds a run can blow is tracked here with its ceiling recorded, and
// reaching one is a fact that lands in the run's record rather than a run that simply
// stopped. Concurrency is the odd one out: it never stops a run, it bounds it, and what
// is recorded is that the bound was saturated, because "the run was limited to two
// requests in flight" changes what a throughput number means.
//
// The four stopping bounds all end the run the same way: the driver unwinds, the cases
// that were never driven are recorded as unavailable, and what was already measured keeps
// its verdicts. A ceiling is an operational fact, so it appears in the notes and in the
// ceilings report. It is not a verdict about the model and it never reaches a score.

import (
	"context"
	"fmt"
	"sync"
	"time"
)

// guardrails is one run's counters. It is safe for the driver's worker goroutines to share.
type guardrails struct {
	limits Ceilings
	sem    chan struct{}
	now    func() time.Time

	mu            sync.Mutex
	inFlight      int
	peak          int
	saturated     bool
	requests      int
	outputBytes   int64
	artifactBytes int64
	artifactKeys  map[string]bool
	hits          []CeilingHit
	// stopped names the bound that ended the run, or "cancelled" when an operator did.
	stopped  string
	deadline time.Time
	cancel   func()
}

// newGuardrails bounds a run. The concurrency of the benchmark version and the run's
// ceiling are both honoured, which means the smaller of them, and the one that was smaller
// is recorded so a reader knows which bound the numbers were taken under.
func newGuardrails(ceilings Ceilings, versionConcurrency int, at time.Time, cancel func()) (*guardrails, int) {
	concurrency := versionConcurrency
	if ceilings.Concurrency < concurrency {
		concurrency = ceilings.Concurrency
	}
	if concurrency < 1 {
		concurrency = 1
	}
	g := &guardrails{
		limits:       ceilings,
		sem:          make(chan struct{}, concurrency),
		now:          time.Now,
		artifactKeys: map[string]bool{},
		deadline:     at.Add(ceilings.WallClock),
		cancel:       cancel,
	}
	return g, concurrency
}

// report returns the ceiling report the run records: the ceilings themselves, the
// effective concurrency, whether it was saturated, and every hit.
func (g *guardrails) report(effectiveConcurrency int) CeilingReport {
	g.mu.Lock()
	defer g.mu.Unlock()
	hits := append([]CeilingHit(nil), g.hits...)
	return CeilingReport{
		Ceilings:             g.limits,
		Hits:                 hits,
		EffectiveConcurrency: effectiveConcurrency,
		PeakInFlight:         g.peak,
		Saturated:            g.saturated,
	}
}

// stoppedBy is the bound that ended the run, empty while it is still going.
func (g *guardrails) stoppedBy() string {
	g.mu.Lock()
	defer g.mu.Unlock()
	return g.stopped
}

// counts is what the run has spent so far.
func (g *guardrails) counts() (requests int, output, artifacts int64) {
	g.mu.Lock()
	defer g.mu.Unlock()
	return g.requests, g.outputBytes, g.artifactBytes
}

// hit records one bound the run reached and stops the run when the bound is a stopping
// one. The first hit is the one the run records as its cause: a run that blew three
// ceilings at once stopped at the first one it was refused at, and saying so is more
// useful than a list in arrival order.
func (g *guardrails) hit(name string, ceiling string, observed string, stops bool) *CeilingHit {
	g.mu.Lock()
	h := CeilingHit{
		Name: name, Ceiling: ceiling, Observed: observed, At: g.now(),
		Detail: fmt.Sprintf("the run stopped there"),
	}
	if !stops {
		h.Detail = "the run was held to that bound and continued"
	}
	g.hits = append(g.hits, h)
	if stops && g.stopped == "" {
		g.stopped = name
	}
	g.mu.Unlock()
	if stops {
		// The cancel is called outside the lock: it wakes the driver, and the driver may
		// come straight back in to ask for a slot.
		if g.cancel != nil {
			g.cancel()
		}
	}
	return &h
}

// slot takes one of the run's in-flight places, and reports whether the run may still
// make a request at all. A ceiling that was already reached is refused here rather than
// after the request, so a stopped run spends no further traffic. Waiting for a place ends
// when the run's own context ends: a cancelled run does not queue work it will never do.
func (g *guardrails) slot(ctx context.Context) (release func(), ok bool) {
	g.mu.Lock()
	if g.stopped != "" {
		g.mu.Unlock()
		return nil, false
	}
	g.mu.Unlock()

	select {
	case g.sem <- struct{}{}:
	case <-ctx.Done():
		return nil, false
	}

	g.mu.Lock()
	g.inFlight++
	if g.inFlight > g.peak {
		g.peak = g.inFlight
	}
	if g.inFlight >= cap(g.sem) && !g.saturated {
		g.saturated = true
		// Concurrency is a bound, not a stop: saturation is recorded so a reader knows
		// the run never got past it, which is the difference between "the endpoint served
		// two at a time" and "the run only ever asked for two".
		g.hits = append(g.hits, CeilingHit{
			Name: CeilingConcurrency, Ceiling: fmt.Sprintf("%d", cap(g.sem)),
			Observed: fmt.Sprintf("%d in flight", g.inFlight), At: g.now(),
			Detail: "the run was held to that bound and continued",
		})
	}
	g.mu.Unlock()

	released := false
	return func() {
		g.mu.Lock()
		if !released {
			released = true
			g.inFlight--
		}
		g.mu.Unlock()
		<-g.sem
	}, true
}

// requestSpend counts one request against the request ceiling and reports the hit when the
// run has used them all up.
func (g *guardrails) requestSpend() *CeilingHit {
	g.mu.Lock()
	if g.stopped != "" {
		g.mu.Unlock()
		return nil
	}
	g.requests++
	used := g.requests
	g.mu.Unlock()
	if used > g.limits.Requests {
		g.mu.Lock()
		g.requests--
		g.mu.Unlock()
		// The observed value is the request the run was refused, not the ceiling: "it asked
		// for the thirty-first" is the fact, and the counter stays at what was actually spent.
		return g.hit(CeilingRequests, fmt.Sprintf("%d requests", g.limits.Requests),
			fmt.Sprintf("%d requests asked for", used), true)
	}
	return nil
}

// outputSpend counts response bytes the run read.
func (g *guardrails) outputSpend(n int64) *CeilingHit {
	g.mu.Lock()
	g.outputBytes += n
	total := g.outputBytes
	g.mu.Unlock()
	if total > g.limits.OutputBytes {
		return g.hit(CeilingOutputBytes, fmt.Sprintf("%d bytes", g.limits.OutputBytes),
			fmt.Sprintf("%d bytes read", total), true)
	}
	return nil
}

// artifactSpend counts bytes the run asked the artifact store to hold. Bytes already
// stored under the same key are not spent twice, because the store holds them once.
func (g *guardrails) artifactSpend(key string, n int64) *CeilingHit {
	g.mu.Lock()
	if !g.artifactKeys[key] {
		g.artifactKeys[key] = true
		g.artifactBytes += n
	}
	total := g.artifactBytes
	g.mu.Unlock()
	if total > g.limits.DiskBytes {
		return g.hit(CeilingDiskBytes, fmt.Sprintf("%d bytes", g.limits.DiskBytes),
			fmt.Sprintf("%d bytes recorded", total), true)
	}
	return nil
}

// timeSpend records that the wall clock bound ended the run. It is called when the run's
// deadline is what expired, which is the difference between a run that ran out of time and
// a run somebody cancelled.
func (g *guardrails) timeSpend() *CeilingHit {
	if g.stoppedBy() != "" {
		return nil
	}
	return g.hit(CeilingWallClock, g.limits.WallClock.String(),
		g.now().Sub(g.deadline.Add(-g.limits.WallClock)).String(), true)
}

// markCancelled says an operator stopped the run. A cancellation that arrives after a
// ceiling does not overwrite the ceiling: the bound is what stopped it, and a run gets one
// recorded cause.
func (g *guardrails) markCancelled() {
	g.mu.Lock()
	defer g.mu.Unlock()
	if g.stopped == "" {
		g.stopped = "cancelled"
	}
}
