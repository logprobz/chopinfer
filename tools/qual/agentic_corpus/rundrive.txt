package bench

// Driving one run: the documents that say what it was pinned to, the cases it drives, the
// attempts it makes, and the account it ends with.
//
// The order of this file is the order of a run. Everything that describes the run is
// written down before a single request goes out, so a run that got no answer at all still
// says which deployment, which version, which ceilings and which cases it was measuring.
// Then the cases are driven under the guard rails, one attempt at a time, and each attempt
// is recorded as it ends rather than at the end of the run: a run that stops halfway leaves
// a complete account of the half it did.
//
// Two rules are structural rather than advisory here.
//
// A pass needs an answer. A verdict of pass is only let stand for an attempt whose exchange
// produced a payload, whoever the judge was: a scorer that returned pass over a request
// that never arrived has its claim refused and rewritten as blocked, and an evaluator
// child that exited 0 while claiming a pass over that same exchange is recorded as having
// made a claim the station refused. Neither the exit code nor the claim reaches the score.
//
// Nothing is written down before it goes through the run's secret set. The raw payload, the
// attempt record, the traffic observation and the report all pass through the same scrub,
// and the attempt record counts the replacements it made rather than pretending the bytes
// never arrived.

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"os"
	"path/filepath"
	"strings"
	"sync"
	"syscall"
	"time"

	"runyard.local/runyard/internal/store"
)

// writeDocument stores one JSON document as one of the run's artifacts and names it in the
// run's evidence. A document that cannot be canonicalised is an error: a run record that
// silently omitted a document would be a record with a hole in it.
func (rs *runState) writeDocument(ctx context.Context, s *RunService, kind string, doc any) (string, int64, error) {
	canon, err := canonicalJSON(doc)
	if err != nil {
		return "", 0, err
	}
	return rs.writeBytes(ctx, s, kind, canon, "application/json")
}

// writeBytes stores bytes as one of the run's artifacts, spends them against the disk
// ceiling, and names the artifact as the run's evidence.
func (rs *runState) writeBytes(ctx context.Context, s *RunService, kind string, body []byte, media string) (string, int64, error) {
	art, err := s.arts.PutBytes(rs.recordCtx, rs.spec.Actor, kind, media, body)
	if err != nil {
		return "", 0, fmt.Errorf("bench: run %s could not record its %s: %w", rs.id, kind, err)
	}
	if hit := rs.guards.artifactSpend(art.Key, art.Size); hit != nil {
		rs.recordHit(hit)
	}
	if err := rs.nameEvidence(s, art.Key, kind); err != nil {
		return art.Key, art.Size, err
	}
	return art.Key, art.Size, nil
}

// nameEvidence records that an artifact is this run's evidence. A pinned run asks the
// artifact store itself, because a pin is the store's own decision with an actor and a time
// on it; a run that pins nothing writes the evidence row directly.
func (rs *runState) nameEvidence(s *RunService, key, kind string) error {
	if !rs.spec.PinEvidence {
		return recordEvidence(rs.recordCtx, s.db, rs.spec.Actor, rs.id, key, kind, s.now())
	}
	err := s.arts.Pin(rs.recordCtx, key, rs.spec.Actor, rs.id, kind)
	if err == nil {
		return nil
	}
	// Content addressing means the same bytes may already be pinned as another run's
	// evidence. That is not a reason to lose this run's claim on them, so the claim is
	// recorded as an ordinary evidence row and the situation is said out loud.
	if pinErr := recordEvidence(rs.recordCtx, s.db, rs.spec.Actor, rs.id, key, kind, s.now()); pinErr != nil {
		return pinErr
	}
	rs.warn("artifact %s was already pinned as another run's evidence; this run names it without pinning it again", key)
	return nil
}

// warn records something an operator needs to know that is not a verdict.
func (rs *runState) warn(format string, args ...any) {
	rs.mu.Lock()
	defer rs.mu.Unlock()
	rs.warnings = append(rs.warnings, fmt.Sprintf(format, args...))
}

// note records one operational fact about the run.
func (rs *runState) note(kind, detail string, at time.Time) {
	rs.mu.Lock()
	defer rs.mu.Unlock()
	rs.notes = append(rs.notes, OperationNote{Kind: kind, Detail: detail, At: at})
	rs.detail.Aggregate.OperationalFailures++
}

// recordHit folds one ceiling hit into the run's ceiling report and notes it. Concurrency
// saturation is the run being held to a bound; every other hit has already stopped it.
func (rs *runState) recordHit(hit *CeilingHit) {
	if hit == nil {
		return
	}
	rs.mu.Lock()
	seen := false
	for _, h := range rs.detail.Ceilings.Hits {
		if h.Name == hit.Name && h.Observed == hit.Observed {
			seen = true
		}
	}
	if !seen {
		rs.detail.Ceilings.Hits = append(rs.detail.Ceilings.Hits, *hit)
	}
	rs.mu.Unlock()
	rs.note(NoteCeiling, hit.String(), hit.At)
}

// snapshot copies the run's warnings and its operational notes as they stand.
func (rs *runState) snapshot() (warnings []string, notes []OperationNote) {
	rs.mu.Lock()
	defer rs.mu.Unlock()
	return append([]string(nil), rs.warnings...), append([]OperationNote(nil), rs.notes...)
}

// recordHardware writes the hardware snapshot the numbers were taken on. A run that was
// handed a reference uses it as it stands; a run handed a document gets a digest of the
// canonical bytes, and those bytes are what the run keeps.
func (rs *runState) recordHardware(ctx context.Context, s *RunService) error {
	pin := rs.spec.Pins.Hardware
	if len(pin.Document) == 0 {
		if pin.Reference == "" {
			return fmt.Errorf("bench: run %s has no hardware snapshot to pin", rs.id)
		}
		rs.detail.Documents[documentHardware] = pin.Reference
		return nil
	}
	key, _, err := rs.writeBytes(ctx, s, ArtifactKindHardware, pin.Document, "application/json")
	if err != nil {
		return err
	}
	// The artifact key is the digest of the canonical bytes the store holds, which is what
	// the pin wants. It is only set after the bytes are on disk: a pin naming a digest the
	// station does not have would be a claim about a file nobody wrote.
	rs.spec.Pins.Hardware.Reference = key
	rs.detail.Documents[documentHardware] = key
	return nil
}

// recordPins writes the pin set as the run's own document. It is written before anything is
// driven so that the first thing a run records about itself is what its numbers mean.
func (rs *runState) recordPins(ctx context.Context, s *RunService) error {
	canon, err := rs.spec.Pins.canonical()
	if err != nil {
		return err
	}
	digest, err := rs.spec.Pins.Digest()
	if err != nil {
		return err
	}
	key, _, err := rs.writeBytes(ctx, s, ArtifactKindPins, canon, "application/json")
	if err != nil {
		return err
	}
	rs.mu.Lock()
	rs.detail.Documents[documentPins] = key
	rs.detail.PinsDigest = digest
	rs.mu.Unlock()
	return nil
}

// observeTraffic asks what else the endpoint was carrying and records the answer, busy or
// not. A probe that could not answer is recorded as having failed rather than dropped: not
// knowing whether the endpoint was shared is a condition of the run's numbers.
func (rs *runState) observeTraffic(ctx context.Context, s *RunService) error {
	window := rs.spec.TrafficWindow
	if window <= 0 {
		window = DefaultTrafficWindow
	}
	probe := rs.spec.Traffic
	if probe == nil {
		probe = s.traffic
	}
	var (
		obs TrafficObservation
		err error
	)
	switch {
	case probe != nil:
		obs, err = probe(ctx)
		obs.ObservedAt = s.now()
		if obs.WindowNanos == 0 {
			obs.WindowNanos = int64(window)
		}
	default:
		obs, err = stationTraffic(ctx, s.db, s.now().Add(-window), rs.spec.Pins.Deployment.ID)
		if err == nil {
			obs.ObservedAt = s.now()
			obs.WindowNanos = int64(window)
			obs.Source = "the station's own request history"
		}
	}
	if err != nil {
		obs.ProbeFailed = err.Error()
		if obs.Source == "" {
			obs.Source = "unknown"
		}
	}
	rs.traffic = &obs
	key, _, werr := rs.writeDocument(ctx, s, ArtifactKindTraffic, obs)
	if werr == nil {
		rs.mu.Lock()
		rs.detail.Documents[documentTraffic] = key
		rs.mu.Unlock()
	}
	rs.mu.Lock()
	rs.detail.Traffic = &obs
	rs.mu.Unlock()
	if err != nil {
		return err
	}
	if werr != nil {
		return werr
	}
	if obs.Busy(rs.spec.WarnConcurrentRequests) {
		rs.warn("the endpoint was carrying other traffic while this run was measured: %d requests in flight and %d finished in the last %s from %s; the run's isolation scope is %q",
			obs.InFlight, obs.RecentRequests, window, strings.Join(obs.OtherDeployments, ", "), rs.spec.Isolation)
	}
	return nil
}

// stationTraffic is the default probe: what the station itself was serving. It answers for
// traffic that goes through the station's own proxy, which is why a caller may supply a
// probe instead; a deployment driven directly is outside what this table can see.
func stationTraffic(ctx context.Context, db *store.Store, since time.Time, exclude string) (TrafficObservation, error) {
	counts, err := db.BenchTraffic(ctx, since, exclude)
	if err != nil {
		return TrafficObservation{}, err
	}
	return TrafficObservation{
		InFlight:         counts.InFlight,
		RecentRequests:   counts.Recent,
		OtherDeployments: counts.OtherDeployments,
	}, nil
}

// stagePackage gets the run's benchmark code onto disk through the approval boundary, and
// only through it. RunSpec has no field that names a directory of code, so there is no
// other route: a run that pins a package either passes the approval check or does not run.
func (rs *runState) stagePackage(ctx context.Context, s *RunService) error {
	pin := rs.spec.Pins.Package
	if pin.NoneStated {
		rs.note(NoteCapability, "the run states that it executes no package code", s.now())
		return nil
	}
	if s.pkgs == nil {
		return fmt.Errorf("bench: run %s pins package %s, so it needs the package approval boundary; this run service was built without one", rs.id, pin.Key)
	}
	root := filepath.Join(s.workspace, "runs", rs.id, "package")
	if err := os.MkdirAll(filepath.Dir(root), 0o700); err != nil {
		return fmt.Errorf("bench: run %s could not make its staging directory: %w", rs.id, err)
	}
	staged, err := s.pkgs.StageForRun(ctx, rs.spec.Actor, s.now(), pin.Key, pin.Digest, pin.Benchmark, root)
	if err != nil {
		return fmt.Errorf("bench: the package approval boundary refused run %s: %w", rs.id, err)
	}
	rs.staged = &staged
	return nil
}

// generateCases produces the cases the sealed version fixed and records every one of them.
// The generator is asked under the version's own seed, count and context slot; a generator
// that answers with a different number of cases than it was asked for is reported rather
// than quietly truncated, because a run that dropped cases would report a smaller
// measurement than it made.
func (rs *runState) generateCases(ctx context.Context, s *RunService) error {
	vs := rs.version.Spec()
	cases, err := rs.gen.Generate(ctx, GenerateRequest{
		Seed:          vs.Inputs.Seed,
		Count:         vs.Inputs.CaseCount,
		ContextTokens: vs.Inputs.ContextTokens,
		Params:        vs.Inputs.Params,
	})
	if err != nil {
		return fmt.Errorf("bench: generator %s could not produce the cases benchmark version %s fixed: %w", vs.Inputs.Generator, rs.version.ID(), err)
	}
	if len(cases) != vs.Inputs.CaseCount {
		return fmt.Errorf("bench: generator %s returned %d cases for the %d that benchmark version %s fixed",
			vs.Inputs.Generator, len(cases), vs.Inputs.CaseCount, rs.version.ID())
	}
	seen := map[string]bool{}
	keys := make([]string, 0, len(cases))
	for _, c := range cases {
		if c.ID == "" {
			return fmt.Errorf("bench: generator %s returned a case with no id", vs.Inputs.Generator)
		}
		if c.Task != nil && (c.Task.Kind == "" || c.Task.Version < 1 || !json.Valid(c.Task.Payload)) {
			return fmt.Errorf("bench: case %s has an invalid task contract", c.ID)
		}
		if seen[c.ID] {
			return fmt.Errorf("bench: generator %s returned case %s twice", vs.Inputs.Generator, c.ID)
		}
		seen[c.ID] = true
		key, canon, err := caseKeyOf(c)
		if err != nil {
			return err
		}
		promptKey, _, err := rs.writeBytes(ctx, s, ArtifactKindCase, canon, "application/json")
		if err != nil {
			return err
		}
		params, err := canonicalJSON(c.Params)
		if err != nil {
			return err
		}
		if err := recordCase(rs.recordCtx, s.db, rs.spec.Actor, s.now(), rs.version.ID(), c, key, promptKey, params); err != nil {
			return err
		}
		keys = append(keys, key)
	}
	rs.cases = cases
	rs.caseKeys = keys

	// Two documents, because a reader wants both halves: the keys in the run's own order,
	// which is what the case rows are read back by, and the digest they fold into, which is
	// what the input hash covers.
	key, _, err := rs.writeDocument(ctx, s, ArtifactKindCaseKeys, keys)
	if err != nil {
		return err
	}
	setDigest := caseSetDigest(keys)
	if _, _, err := rs.writeDocument(ctx, s, ArtifactKindCaseDigests, struct {
		CaseSet string   `json:"caseSet"`
		Keys    []string `json:"keys"`
	}{CaseSet: setDigest, Keys: keys}); err != nil {
		return err
	}
	rs.mu.Lock()
	rs.detail.Documents[documentCaseKeys] = key
	rs.detail.Documents[documentCaseDigest] = setDigest
	rs.detail.Aggregate.CasesPlanned = len(cases) * repetitions(rs.stored)
	rs.mu.Unlock()

	inputsHash, err := rs.spec.Pins.InputsHash(setDigest)
	if err != nil {
		return err
	}
	rs.mu.Lock()
	rs.detail.InputsHash = inputsHash
	rs.mu.Unlock()
	return nil
}

// repetitions is how many times the sealed version drove each case.
func repetitions(stored StoredVersion) int {
	if stored.Spec.Repetitions < 1 {
		return 1
	}
	return stored.Spec.Repetitions
}

// stoppedReason is the bound or the stop that ended the run, empty while it is still going.
func (rs *runState) stoppedReason() string { return rs.guards.stoppedBy() }

// mayDrive says whether the run may start another attempt: a ceiling, a cancellation or an
// expired context all end the driving, and each of them is already recorded.
func (rs *runState) mayDrive(ctx context.Context) bool {
	if rs.guards.stoppedBy() != "" {
		return false
	}
	select {
	case <-ctx.Done():
		return false
	default:
		return true
	}
}

// drive drives every case the run planned, under the guard rails. Cases it could not start
// are counted as unavailable rather than driven into a verdict they did not earn.
func (rs *runState) drive(ctx context.Context, s *RunService) {
	if rs.spec.Agentic != nil {
		rs.driveAgentic(ctx, s)
		return
	}
	reps := repetitions(rs.stored)
	type job struct {
		index int
		rep   int
	}
	jobs := make([]job, 0, len(rs.cases)*reps)
	for rep := 1; rep <= reps; rep++ {
		for i := range rs.cases {
			jobs = append(jobs, job{index: i, rep: rep})
		}
	}

	stopCtx, stopWatch := context.WithCancel(ctx)
	defer stopWatch()
	go rs.watchStop(stopCtx, s)
	go rs.watchDeadline(stopCtx, s)

	limit := rs.concurrency
	if limit < 1 {
		limit = 1
	}
	var (
		mu       sync.Mutex
		next     int
		driven   = make([]bool, len(jobs))
		workers  sync.WaitGroup
		complete int
	)
	for w := 0; w < limit; w++ {
		workers.Add(1)
		go func() {
			defer workers.Done()
			for {
				mu.Lock()
				i := next
				if i < len(jobs) {
					next++
				}
				mu.Unlock()
				if i >= len(jobs) {
					return
				}
				if !rs.mayDrive(ctx) {
					return
				}
				if rs.driveOne(ctx, s, jobs[i].index, jobs[i].rep) {
					mu.Lock()
					driven[i] = true
					complete++
					every := rs.spec.ProgressEvery
					if every < 1 {
						every = 1
					}
					emit := complete%every == 0
					mu.Unlock()
					// The progress is written whether or not anyone is watching from
					// inside this process: a dashboard reading the rows is the reader this
					// is for, and a callback is only one of its consumers.
					if emit {
						rs.emitProgress(ctx, s)
					}
				}
			}
		}()
	}
	workers.Wait()

	rs.mu.Lock()
	drivenCount := 0
	for _, ok := range driven {
		if ok {
			drivenCount++
		}
	}
	rs.detail.Aggregate.CasesDriven = drivenCount
	rs.detail.Aggregate.Unavailable = len(jobs) - drivenCount
	unavailable := rs.detail.Aggregate.Unavailable
	rs.mu.Unlock()
	if unavailable > 0 {
		rs.note(NoteUnavailable, fmt.Sprintf("%d of %d planned attempts were never driven, and they are recorded as unavailable rather than as zeros",
			unavailable, len(jobs)), s.now())
		rs.warn("%d of %d planned attempts were never driven", unavailable, len(jobs))
	}
	rs.emitProgress(ctx, s)
}

// driveAgentic keeps C coding sessions occupied during the fixed interval. The
// ordinary attempt path still owns admission, evidence, verdicts and ceilings.
func (rs *runState) driveAgentic(ctx context.Context, s *RunService) {
	stopCtx, stopWatch := context.WithCancel(ctx)
	defer stopWatch()
	go rs.watchStop(stopCtx, s)
	go rs.watchDeadline(stopCtx, s)
	plan := *rs.spec.Agentic
	onWindow := plan.OnWindow
	plan.OnWindow = func(start, end time.Time) {
		rs.mu.Lock()
		rs.agenticWindowEnd = end
		rs.mu.Unlock()
		if onWindow != nil {
			onWindow(start, end)
		}
	}
	drive := func(ctx context.Context, index int, phase string) (AgenticSessionTrace, error) {
		if index >= len(rs.cases) {
			return AgenticSessionTrace{}, fmt.Errorf("agentic: case plan exhausted")
		}
		id := rs.cases[index].ID
		rs.mu.Lock()
		rs.agenticPhases[id] = phase
		rs.mu.Unlock()
		if !rs.mayDrive(ctx) {
			return AgenticSessionTrace{}, fmt.Errorf("agentic: run stopped")
		}
		if !rs.driveOne(ctx, s, index, 1) {
			return AgenticSessionTrace{}, fmt.Errorf("agentic: case %s was not recorded", id)
		}
		rs.mu.Lock()
		rec, ok := rs.agenticRecords[id]
		rs.mu.Unlock()
		if !ok {
			return AgenticSessionTrace{}, fmt.Errorf("agentic: case %s has no attempt record", id)
		}
		if rec.Response.Agentic == nil {
			return AgenticSessionTrace{StartedAt: rec.StartedAt, FinishedAt: rec.FinishedAt, Failure: rec.Failure}, nil
		}
		return *rec.Response.Agentic, nil
	}
	report, _, err := DriveAgenticLevel(ctx, plan, drive)
	if err != nil {
		rs.warn("agentic level stopped: %v", err)
	}
	rs.mu.Lock()
	rs.detail.AgenticReport = &report
	rs.detail.Aggregate.CasesPlanned = report.MeasuredCompleted
	rs.detail.Aggregate.Unavailable = 0
	rs.mu.Unlock()
	rs.emitProgress(ctx, s)
}

// watchStop polls the durable cancellation flag, so a run driven by one process is stopped
// by another that only has the database.
func (rs *runState) watchStop(ctx context.Context, s *RunService) {
	interval := DefaultCancellationInterval
	ticker := time.NewTicker(interval)
	defer ticker.Stop()
	for {
		select {
		case <-ctx.Done():
			return
		case <-ticker.C:
			want, err := cancelRequested(ctx, s.db, rs.id)
			if err != nil || !want {
				if err != nil && !errors.Is(err, context.Canceled) {
					s.logf("bench: run %s could not read its cancellation: %v", rs.id, err)
				}
				continue
			}
			rs.guards.markCancelled()
			rs.note(NoteCancelled, "a cancel request reached the run's job record", s.now())
			rs.cancel()
			return
		}
	}
}

// watchDeadline stops the run when its wall clock ceiling expires, recording the ceiling
// with the time the run had actually taken.
func (rs *runState) watchDeadline(ctx context.Context, s *RunService) {
	timer := time.NewTimer(time.Until(rs.deadline))
	defer timer.Stop()
	select {
	case <-ctx.Done():
		return
	case <-timer.C:
		rs.recordHit(rs.guards.timeSpend())
		rs.cancel()
	}
}

// emitProgress writes the run's progress into its detail document and hands a copy to the
// caller's callback, if it set one. The numbers come from the attempts as they were recorded
// and from the same rows any other reader would read, so the callback is a convenience
// rather than the source of truth, and a reader that never saw the callback still sees the
// run move.
func (rs *runState) emitProgress(ctx context.Context, s *RunService) {
	rs.mu.Lock()
	records := append([]CaseRecord(nil), rs.records...)
	tallyRun(rs.version.ID(), &rs.detail.Aggregate, records, nil)
	rs.detail.Ceilings = rs.guards.report(rs.concurrency)
	detail := rs.detail
	rs.mu.Unlock()
	if err := setRunState(rs.recordCtx, s.db, rs.spec.Actor, rs.id, RunStateRunning, "", detail, s.now()); err != nil {
		s.logf("bench: run %s could not record its progress: %v", rs.id, err)
		return
	}
	if rs.spec.Progress == nil {
		return
	}
	progress, err := readProgress(rs.recordCtx, s.db, rs.id)
	if err != nil {
		return
	}
	rs.spec.Progress(progress)
}

// tallyRun folds the attempts recorded so far into the run's aggregate. A run that stopped
// early keeps the verdicts it reached and accounts for the rest as unavailable, so the
// numbers always add up to the plan and never to a score the run did not measure.
//
// The score is the fraction of *measured* verdicts that passed. Blocked and inapplicable
// verdicts are counted and named, never folded into a denominator that would let a run of
// transport failures read as a run of low scores.
func tallyRun(versionID string, agg *RunAggregate, records []CaseRecord, planned *int) {
	if planned != nil {
		agg.CasesPlanned = *planned
	}
	agg.Passes, agg.Failures, agg.Inapplicable, agg.Blocked = 0, 0, 0, 0
	agg.TransportFailures, agg.MeasuredZero = 0, 0
	driven := 0
	for _, rec := range records {
		if rec.Phase == "warmup" || rec.Phase == "drain" {
			continue
		}
		if len(rec.Verdicts) > 0 {
			driven++
		}
		if rec.Transport == TransportFailed {
			agg.TransportFailures++
		}
		if rec.Score != nil && *rec.Score == 0 {
			agg.MeasuredZero++
		}
		for _, v := range rec.Verdicts {
			switch v.State() {
			case StatePass:
				agg.Passes++
			case StateFail:
				agg.Failures++
			case StateInapplicable:
				agg.Inapplicable++
			case StateBlocked:
				agg.Blocked++
			}
		}
	}
	agg.CasesDriven = driven
	if agg.CasesPlanned > 0 {
		agg.Coverage = float64(driven) / float64(agg.CasesPlanned)
	}
	switch measured := agg.Passes + agg.Failures; {
	case measured == 0:
		agg.Score = nil
		agg.ScoreUnavailable = fmt.Sprintf(
			"no case produced a measured verdict: %d blocked, %d inapplicable, %d of %d attempts never driven",
			agg.Blocked, agg.Inapplicable, agg.Unavailable, agg.CasesPlanned)
	default:
		score := float64(agg.Passes) / float64(measured)
		agg.Score = &score
		agg.ScoreUnavailable = ""
	}
}

// driveOne drives one case once, records the attempt, and reports whether the attempt
// reached a verdict. It returns false when the run had already been stopped and nothing was
// attempted, which is what keeps an unstarted attempt out of the counts.
func (rs *runState) driveOne(ctx context.Context, s *RunService, index, rep int) bool {
	c := rs.cases[index]
	target := rs.spec.Target
	at := s.now()
	rec := CaseRecord{
		RunID: rs.id, CaseID: c.ID, Repetition: rep,
		Executor: string(rs.execID), ScorerIDs: rs.scorerIDs(),
	}
	rs.mu.Lock()
	rec.Phase = rs.agenticPhases[c.ID]
	rs.mu.Unlock()

	caseCtx, cancelCase := context.WithTimeout(ctx, rs.caseTimeoutOrUnbounded())
	defer cancelCase()

	// A case that needs something the target never declared is not driven at all. The
	// attempt is still recorded, because "we never asked it" is a fact about the run that a
	// reader needs in order to interpret the missing number.
	needs := rs.needsFor(c)
	if missing := needs.StatedByTarget().Missing(target.Declared); len(missing) > 0 {
		rec.Transport = TransportNotAttempted
		for _, sc := range rs.scorers {
			rec.Verdicts = append(rec.Verdicts, InapplicableVerdict(missing[0], fmt.Sprintf(
				"target %s does not declare %s, which case %s needs and scorer %s judges",
				target.ID, missing[0], c.ID, sc.id)))
		}
		return rs.recordAttempt(ctx, s, rec, at)
	}

	release, ok := rs.guards.slot(ctx)
	if !ok {
		return false
	}
	defer release()
	if rs.spec.Agentic == nil {
		if hit := rs.guards.requestSpend(); hit != nil {
			rs.recordHit(hit)
			return false
		}
	}

	// The admission row is written before the request is written to the wire. A station that
	// died in between has a record of a request whose outcome nobody knows, which is the
	// only honest thing it can say.
	no, err := admitAttempt(rs.recordCtx, s.db, rs.spec.Actor, rs.id, rs.spec.Pins.Deployment, rec, s.now())
	if err != nil {
		s.logf("bench: run %s could not admit an attempt for case %s: %v", rs.id, c.ID, err)
		rs.note(NoteRefusal, fmt.Sprintf("case %s could not be admitted: %v", c.ID, err), s.now())
		return false
	}
	rec.AttemptNo = no
	rec.StartedAt = s.now()

	req := ExecuteRequest{
		Case:             c,
		Package:          rs.staged,
		MaxResponseBytes: rs.version.Spec().Budgets.MaxResponseBytes,
		Target:           target,
		Params:           rs.version.Spec().Requests,
		Timeout:          rs.requestTimeout,
		Headers:          rs.spec.Headers,
		Client:           rs.client,
	}
	if rs.spec.Agentic != nil {
		req.BeforeRequest = func() error {
			if !rs.mayDrive(caseCtx) {
				return fmt.Errorf("agentic: run stopped before next HTTP turn")
			}
			if hit := rs.guards.requestSpend(); hit != nil {
				rs.recordHit(hit)
				return fmt.Errorf("agentic: request ceiling reached")
			}
			return nil
		}
		req.AfterResponse = func(bytes int64) error {
			if hit := rs.guards.outputSpend(bytes); hit != nil {
				rs.recordHit(hit)
				return fmt.Errorf("agentic: response-byte ceiling reached")
			}
			return nil
		}
	}
	attempt, err := rs.executor.Execute(caseCtx, req)
	rec.Driven = true
	if err != nil {
		// The executor could not produce an attempt at all. That is an exchange that did
		// not happen, recorded as one; it is never a verdict about the model.
		rec.Transport = TransportFailed
		rec.Failure = err.Error()
		rec.Notes = append(rec.Notes, OperationNote{Kind: NoteTransport, Detail: err.Error(), At: s.now()})
		for _, sc := range rs.scorers {
			rec.Verdicts = append(rec.Verdicts, BlockedVerdict(Need{What: NeedEndpoint}, ReasonTransport,
				fmt.Sprintf("the executor for %s could not drive the case: %v", sc.id, err)))
		}
		return rs.recordAttempt(ctx, s, rec, at)
	}

	rec.ExecutorDescriptor = attempt.Executor
	rec.Observed = attempt.Observed
	rec.Failure = attempt.Observed.Failure()
	switch rec.Failure {
	case "":
		rec.Transport = TransportAnswered
	default:
		rec.Transport = TransportFailed
		rec.Notes = append(rec.Notes, OperationNote{Kind: NoteTransport,
			Detail: rec.Failure, At: s.now()})
	}
	if rs.spec.Agentic == nil {
		if hit := rs.guards.outputSpend(int64(attempt.Observed.ResponseBytes)); hit != nil {
			rs.recordHit(hit)
		}
	}

	// The payload is kept as it arrived, minus whatever the operator declared secret. The
	// record says how many replacements were made and how big the bytes were before, so a
	// reader knows the retained artifact is not byte-identical and by how much.
	raw := attempt.Response.Raw
	scrubbed, redactions := rs.secrets.Scrub(raw)
	rec.Evidence.Redactions = redactions
	rec.Evidence.RedactedFrom = int64(len(raw))
	if len(raw) > 0 {
		key, size, err := rs.writeBytes(ctx, s, ArtifactKindRaw, scrubbed, "application/json")
		if err != nil {
			s.logf("bench: run %s could not keep the payload of case %s: %v", rs.id, c.ID, err)
			rs.note(NoteRefusal, fmt.Sprintf("the raw payload of case %s could not be retained: %v", c.ID, err), s.now())
		} else {
			rec.Evidence.RawArtifactKey = key
			rec.Evidence.RawBytes = size
		}
		if redactions > 0 {
			rec.Notes = append(rec.Notes, OperationNote{Kind: NoteRedaction, At: s.now(),
				Detail: fmt.Sprintf("%d declared secret value(s) were replaced in %d bytes of retained payload", redactions, len(raw))})
		}
	}
	// The row holds the decoded answer; the raw bytes live once, in the artifact above.
	rec.Response = attempt.Response
	rec.Response.Raw = nil

	for _, sc := range rs.scorers {
		v, err := sc.sc.Score(caseCtx, c, attempt)
		if err != nil {
			// A scorer that could not run is an operational failure, not a case outcome. It
			// is recorded as blocked with the scorer named, so the run does not lose the
			// case silently and does not invent a verdict either.
			rec.Notes = append(rec.Notes, OperationNote{Kind: NoteRefusal, At: s.now(),
				Detail: fmt.Sprintf("scorer %s could not judge case %s: %v", sc.id, c.ID, err)})
			rec.Verdicts = append(rec.Verdicts, BlockedVerdict(Need{What: NeedEndpoint}, ReasonNoKey,
				fmt.Sprintf("scorer %s could not run: %v", sc.id, err)))
			continue
		}
		rec.Verdicts = append(rec.Verdicts, v)
	}

	if rs.spec.Evaluator != nil {
		rs.runEvaluator(caseCtx, s, c, attempt, &rec)
	}

	// The invariant, applied once, after every judge has spoken: a pass needs an answer.
	refuseUnanswerablePasses(&rec)
	rec.FinishedAt = s.now()
	return rs.recordAttempt(ctx, s, rec, at)
}

// recordAttempt finishes one attempt: its record artifact, its row, and its place in the
// run's in-memory account.
func (rs *runState) recordAttempt(ctx context.Context, s *RunService, rec CaseRecord, startedAt time.Time) bool {
	rec.FinishedAt = s.now()
	rs.mu.Lock()
	if rec.Phase == "measured" && !rs.agenticWindowEnd.IsZero() && rec.FinishedAt.After(rs.agenticWindowEnd) {
		rec.Phase = "drain"
	}
	rs.mu.Unlock()
	if rec.StartedAt.IsZero() {
		rec.StartedAt = startedAt
	}
	if rec.AttemptNo == 0 {
		no, err := admitAttempt(rs.recordCtx, s.db, rs.spec.Actor, rs.id, rs.spec.Pins.Deployment, rec, startedAt)
		if err != nil {
			s.logf("bench: run %s could not admit an attempt for case %s: %v", rs.id, rec.CaseID, err)
			return false
		}
		rec.AttemptNo = no
	}
	rec.Score = attemptScore(rec.Verdicts)

	// The record is scrubbed as a whole, not only its payload: an endpoint's refusal body, a
	// verdict detail quoting what came back, and an evaluator's own words are all places a
	// declared secret can reappear. The replacement is recorded rather than silent, so a
	// reader knows the retained record is not byte-identical to what the station saw.
	if raw, err := json.Marshal(rec); err == nil {
		if scrubbed, n := rs.secrets.Scrub(raw); n > 0 {
			var clean CaseRecord
			if err := json.Unmarshal(scrubbed, &clean); err != nil {
				s.logf("bench: run %s could not sanitize the record of case %s: %v", rs.id, rec.CaseID, err)
				return false
			}
			clean.Evidence.Redactions += n
			clean.Evidence.RedactedFrom += int64(len(raw))
			clean.Notes = append(clean.Notes, OperationNote{Kind: NoteRedaction, At: rec.FinishedAt,
				Detail: fmt.Sprintf("%d declared secret value(s) were replaced across the attempt record", n)})
			rec = clean
		}
	}

	// The record artifact is the record as it stood before its own key was known; the row
	// holds the same document with the key added. A document cannot contain its own digest,
	// and saying which copy holds which is what lets a reader check either.
	key, _, err := rs.writeDocument(rs.recordCtx, s, ArtifactKindAttempt, rec)
	if err != nil {
		s.logf("bench: run %s could not record the attempt document of case %s: %v", rs.id, rec.CaseID, err)
	} else {
		rec.Evidence.RecordArtifact = key
	}
	if err := finishAttempt(rs.recordCtx, s.db, rs.spec.Actor, rs.id, rec.AttemptNo, rec, s.now()); err != nil {
		s.logf("bench: run %s could not record attempt %d: %v", rs.id, rec.AttemptNo, err)
		return false
	}
	rs.mu.Lock()
	rs.records = append(rs.records, rec)
	if rs.spec.Agentic != nil {
		rs.agenticRecords[rec.CaseID] = rec
	}
	rs.mu.Unlock()
	return true
}

// refuseUnanswerablePasses is the rule that a transport failure cannot become a model pass.
// It runs after every judge, whether the judge was a registered scorer or an evaluator
// child that exited 0, and it refuses any pass that has no answer behind it. The claim is
// kept as a note next to the verdict that replaced it, so the record shows both what the
// judge said and that the station did not accept it.
//
// It is a function of the record and nothing else, which is what lets the offline verifier
// apply the same rule to the verdicts it re-derives instead of writing a second version of
// it: a verifier with its own copy of a rule verifies its copy.
func refuseUnanswerablePasses(rec *CaseRecord) {
	if rec.Transport == TransportAnswered {
		return
	}
	for i, v := range rec.Verdicts {
		if v.State() != StatePass {
			continue
		}
		rec.Verdicts[i] = BlockedVerdict(Need{What: NeedEndpoint}, transportReason(rec.Observed),
			fmt.Sprintf("a judge claimed a pass over an exchange that did not answer (%s); the claim is recorded and refused: %s",
				rec.Transport, v.Detail()))
		rec.Notes = append(rec.Notes, OperationNote{Kind: NoteClaimRefused, At: rec.FinishedAt,
			Detail: fmt.Sprintf("case %s: a pass claim was refused because the exchange %s: %s",
				rec.CaseID, describeTransport(rec.Transport), v.Detail())})
	}
	if rec.Evaluator != nil && rec.Evaluator.ClaimAccepted {
		rec.Evaluator.ClaimAccepted = false
		rec.Evaluator.ClaimRefused = fmt.Sprintf("the exchange %s, so the child's claim of a pass is not a measurement",
			describeTransport(rec.Transport))
		rec.Notes = append(rec.Notes, OperationNote{Kind: NoteClaimRefused, At: rec.FinishedAt,
			Detail: fmt.Sprintf("case %s: evaluator %s exited %s and claimed a pass over an exchange that %s; the claim was refused",
				rec.CaseID, rec.Evaluator.ID, exitWords(rec.Evaluator), rec.Evaluator.ClaimRefused)})
	}
}

// runEvaluator calls one evaluator child for one attempt and records what it said. The
// child's exit status and its report are recorded together, and neither of them decides the
// attempt's state: an exit code is a fact about a process.
func (rs *runState) runEvaluator(ctx context.Context, s *RunService, c Case, attempt Attempt, rec *CaseRecord) {
	spec := *rs.spec.Evaluator
	dir := filepath.Join(s.workspace, "runs", rs.id, "evaluator")
	cred := filepath.Join(s.workspace, "runs", rs.id, "credentials")
	spec.ArtifactRoot = dir
	spec.CredentialRoot = cred
	for _, d := range []string{dir, cred} {
		if err := os.MkdirAll(d, 0o700); err != nil {
			rec.Notes = append(rec.Notes, OperationNote{Kind: NoteEvaluatorFailed, At: s.now(),
				Detail: fmt.Sprintf("the station could not make the evaluator's directory: %v", err)})
			return
		}
	}
	ev, err := RunEvaluator(ctx, spec, rs.runContext(), s.access, rs.runData())
	record := &EvaluatorAttemptRecord{
		ID: string(spec.ID), Pid: ev.Stats.Pid, ExitStatus: ev.Stats.ExitStatus,
		Terminated: ev.Stats.Terminated, Negotiated: ev.Negotiated, CredentialID: ev.CredentialID,
		Artifacts: ev.Artifacts, At: s.now(),
	}
	record.ExitCode, record.Signal = parseExitStatus(ev.Stats.ExitStatus)
	if len(ev.Report.State) > 0 {
		report := ev.Report
		record.Report = &report
		verdict := ev.Verdict
		record.Verdict = &verdict
	}
	if err != nil {
		record.ClaimRefused = err.Error()
		rec.Notes = append(rec.Notes, OperationNote{Kind: NoteEvaluatorFailed, At: s.now(),
			Detail: fmt.Sprintf("evaluator %s failed: %v (exit %s)", spec.ID, err, exitWords(record))})
	}
	// The child's verdict joins the case's verdicts only when the exchange answered it a
	// payload. A child that reported pass over an exchange that failed is refused here and
	// again in refuseUnanswerablePasses, and its exit status is recorded either way.
	if ev.Verdict.State() != "" && rec.Transport == TransportAnswered && err == nil {
		record.ClaimAccepted = true
		rec.Verdicts = append(rec.Verdicts, ev.Verdict)
	} else if ev.Verdict.State() != "" {
		record.ClaimRefused = fmt.Sprintf("the exchange %s", describeTransport(rec.Transport))
	}
	rec.Evaluator = record
}

// needsFor is everything the case and every scorer judging it require of a target.
func (rs *runState) needsFor(c Case) Needs {
	out := Needs{}
	for _, n := range c.Needs {
		out = out.With(n)
	}
	for _, sc := range rs.scorers {
		for _, n := range sc.sc.Needs(c) {
			out = out.With(n)
		}
	}
	return out
}

// scorerIDs names the judges the attempt was put to.
func (rs *runState) scorerIDs() []string {
	out := make([]string, 0, len(rs.scorers))
	for _, sc := range rs.scorers {
		out = append(out, sc.id.String())
	}
	return out
}

// caseTimeoutOrUnbounded keeps a zero case timeout from meaning "cancel immediately".
func (rs *runState) caseTimeoutOrUnbounded() time.Duration {
	if rs.caseTimeout <= 0 {
		return time.Hour
	}
	return rs.caseTimeout
}

// runContext is what the station tells an evaluator about this run. It is derived from the
// station's own record, never from anything the child says.
func (rs *runState) runContext() RunContext {
	return RunContext{
		RunID: rs.id, BenchmarkVersionID: rs.version.ID(), CaseCount: len(rs.cases),
		Target: rs.spec.Target, StartedAt: rs.startedAt,
	}
}

// runData is what the evaluator callback serves: this run's description, its cases, and the
// attempt records recorded so far. A run id other than this one's is refused, because a
// credential is supposed to read one run and no other.
func (rs *runState) runData() RunData {
	return RunData{
		Context: func(_ context.Context, runID string) (RunContext, error) {
			if runID != rs.id {
				return RunContext{}, fmt.Errorf("bench: run %s has no records for run %s", rs.id, runID)
			}
			return rs.runContext(), nil
		},
		Cases: func(_ context.Context, runID string) ([]Case, error) {
			if runID != rs.id {
				return nil, fmt.Errorf("bench: run %s has no cases for run %s", rs.id, runID)
			}
			return append([]Case(nil), rs.cases...), nil
		},
		Records: func(_ context.Context, runID string) ([]json.RawMessage, error) {
			if runID != rs.id {
				return nil, fmt.Errorf("bench: run %s has no records for run %s", rs.id, runID)
			}
			rs.mu.Lock()
			defer rs.mu.Unlock()
			out := make([]json.RawMessage, 0, len(rs.records))
			for _, rec := range rs.records {
				raw, err := json.Marshal(rec)
				if err != nil {
					return nil, err
				}
				out = append(out, raw)
			}
			return out, nil
		},
	}
}

// attemptScore is the fraction of an attempt's measured verdicts that passed, and nil when
// none of them measured anything.
func attemptScore(verdicts []Verdict) *float64 {
	var pass, measured int
	for _, v := range verdicts {
		if !v.MeasuredState() {
			continue
		}
		measured++
		if v.State() == StatePass {
			pass++
		}
	}
	if measured == 0 {
		return nil
	}
	score := float64(pass) / float64(measured)
	return &score
}

// transportReason names the kind of exchange failure the way a verdict reason does.
func transportReason(o Observation) Reason {
	switch {
	case o.TimedOut:
		return ReasonDeadline
	case o.TransportError != "":
		return ReasonTransport
	case o.Status >= 400:
		return ReasonEndpointRefused
	default:
		return ReasonEndpointRefused
	}
}

// describeTransport finishes the sentence "the exchange ...".
func describeTransport(transport string) string {
	switch transport {
	case TransportAnswered:
		return "answered"
	case TransportFailed:
		return "did not answer"
	case TransportNotAttempted:
		return "was never attempted"
	default:
		return "ended with an outcome the station cannot account for"
	}
}

// parseExitStatus reads the exit code or the signal out of the host's own words for how a
// child ended. Both stay nullable: an absence of information is not exit code zero.
func parseExitStatus(status string) (*int, *int) {
	status = strings.TrimSpace(status)
	if status == "" {
		return nil, nil
	}
	if rest, ok := strings.CutPrefix(status, "exit status "); ok {
		var code int
		if _, err := fmt.Sscanf(rest, "%d", &code); err == nil {
			c := code
			return &c, nil
		}
		return nil, nil
	}
	if rest, ok := strings.CutPrefix(status, "signal "); ok {
		name := strings.TrimSpace(strings.SplitN(rest, " ", 2)[0])
		for i := 1; i <= 64; i++ {
			if sig := syscall.Signal(i); sig.String() == name {
				v := i
				return nil, &v
			}
		}
	}
	return nil, nil
}

// exitWords says how a child ended, in the record's own words.
func exitWords(e *EvaluatorAttemptRecord) string {
	if e == nil {
		return "no exit status"
	}
	switch {
	case e.ExitStatus != "":
		return e.ExitStatus
	case e.ExitCode != nil:
		return fmt.Sprintf("exit code %d", *e.ExitCode)
	case e.Signal != nil:
		return fmt.Sprintf("signal %d", *e.Signal)
	default:
		return "no exit status was observed"
	}
}

// buildReport folds the run's attempt records into the report an exporter takes. Each case
// appears once, carrying every verdict that was reached for it across repetitions and
// scorers; the attempts themselves are retained in full as rows and as artifacts.
func (rs *runState) buildReport(finished time.Time) (Report, error) {
	byCase := map[string][]Verdict{}
	var order []string
	rs.mu.Lock()
	records := append([]CaseRecord(nil), rs.records...)
	rs.mu.Unlock()
	for _, rec := range records {
		if _, seen := byCase[rec.CaseID]; !seen {
			order = append(order, rec.CaseID)
		}
		byCase[rec.CaseID] = append(byCase[rec.CaseID], rec.Verdicts...)
	}
	results := make([]Result, 0, len(order))
	for _, id := range order {
		verdicts := byCase[id]
		if len(verdicts) == 0 {
			// A case with no verdict is not a pass and is not a fail. It is left out of the
			// report and said so in the run's notes, which is where an operational fact
			// belongs.
			rs.note(NoteUnavailable, fmt.Sprintf("case %s produced no verdict and is not in the report", id), finished)
			continue
		}
		results = append(results, Result{CaseID: id, Verdicts: verdicts, At: finished})
	}
	if len(results) == 0 {
		return Report{}, fmt.Errorf("bench: run %s has no verdicts to report", rs.id)
	}
	summary, err := Summarize(results)
	if err != nil {
		return Report{}, err
	}
	warnings, _ := rs.snapshot()
	finishedAt := finished
	if finishedAt.Before(rs.startedAt) {
		finishedAt = rs.startedAt
	}
	return Report{
		Version: rs.version, Target: rs.spec.Target, Results: results, Summary: summary,
		StartedAt: rs.startedAt, FinishedAt: finishedAt, Notes: warnings,
	}, nil
}
