package bench

// The durable run, end to end.
//
// One entry point, Run, does the whole job a CLI and the dashboard will both drive: it
// pins, it refuses what it cannot pin, it puts the run and its job record on disk before
// anything is driven, it generates and records the cases, it drives them through the
// registry's executor and the registry's scorers, it keeps every attempt, every raw
// payload, every transport receipt and every operational failure, and it ends in one of
// the states the runs column knows about. Nothing about a run lives only in memory: the
// progress of a run is a query, its outcome is a row, and its evidence is named by
// run_evidence rows that keep the bytes.
//
// Three rules are structural here rather than polite, and they are the reason this file is
// long:
//
// A transport failure stays a transport failure. The scorers are what turn an attempt into
// a verdict, and their own gate refuses to judge an exchange that never produced a payload,
// so a failed request becomes a blocked verdict on the way in rather than a score. A child
// evaluator's verdict is then *merged* by one rule: a child that claims a pass over an
// exchange that failed is refused, and the claim is recorded as an operational note with
// the child's exit status beside it. An exit code, a signal and a retry counter are all
// recorded, and none of them can decide correctness.
//
// Approved code reaches disk through one call. A run that pins a package gets its staging
// directory from PackageService.StageForRun and there is no field anywhere on RunSpec that
// names a directory of benchmark code, so a caller cannot route around the approval by
// writing a path. A refusal there ends the run and is recorded as a refusal.
//
// Nothing a run is handed in secret is written down. Every byte that reaches a row or an
// artifact goes through the run's SecretSet first.

	"context"

// DefaultExecutorID and DefaultExportID are the built-in ids a run gets when its caller

const (
	DefaultExecutorID = IDChatExecutor
	DefaultExportID   = IDJSONExport

// DefaultCancellationInterval is how often the driver looks at the durable cancellation
// flag, so a run driven by one process can be stopped by another.
const DefaultCancellationInterval = 250 * time.Millisecond

// PackageStager is the approval boundary, seen from the run's side. *PackageService is the
// implementation, and the run service holds an interface rather than a path so that the
// only way to get benchmark code onto disk for a run is to pass the approval check.
type PackageStager interface {
	StageForRun(ctx context.Context, actor string, at time.Time,
		packageKey, digest, benchmark, root string) (StagedPackage, error)
}

// var _ PackageStager = (*PackageService)(nil) is asserted in packageservice's own file
// area below, so a signature drift is a compile error rather than a nil interface at run
// time.

// RunServiceOptions is what a run service is built from.
type RunServiceOptions struct {
	// Packages is the approval boundary. A run that pins a package without it is refused;
	// a run that states it executes no package code does not need it.
	Packages PackageStager
	// Access mints the credentials an evaluator child is given. Required when a run
	// declares an evaluator.
	Access *RunAccess
	// Traffic, when set, is the default probe for what else the endpoint is carrying. A
	// run's own probe wins.
	Traffic TrafficProbe
	// WorkspaceRoot is the station-owned directory under which a run's staging, evaluator
	// scratch and documents live. Empty uses bench/ inside the state directory.
	WorkspaceRoot string
	// Logf, when set, receives the service's own lines. They are written by the same code
	// that writes the record, so they carry no secret the record does not.
	Logf func(format string, args ...any)
}

// RunService drives durable runs over one store.
type RunService struct {
	db        *store.Store
	arts      *store.Artifacts
	reg       *Registries
	pkgs      PackageStager
	access    *RunAccess
	traffic   TrafficProbe
	workspace string
	logf      func(format string, args ...any)

	mu      sync.Mutex
	cancels map[string]context.CancelFunc
	// guards holds the bounds of the runs this process is driving, so a cancellation that
	// arrives here can name itself as the run's stop cause rather than leaving the run to
	// infer it from a cancelled context.
	guards map[string]*guardrails
	now    func() time.Time
}

// NewRunService builds a run service. The registries are the caller's, which is what lets a
// station with its own scorers drive runs with them; the versions found in them are what
// every run records as its observed versions.
func NewRunService(db *store.Store, arts *store.Artifacts, reg *Registries, opts RunServiceOptions) (*RunService, error) {
	if db == nil {
		return nil, fmt.Errorf("bench: a run service needs a store")
	}
	if arts == nil {
		return nil, fmt.Errorf("bench: a run service needs the artifact store, a run's evidence is artifacts")
	}
	if reg == nil {
		return nil, fmt.Errorf("bench: a run service needs the registries it dispatches through")
	}
	workspace := opts.WorkspaceRoot
	if workspace == "" {
		workspace = filepath.Join(db.Paths().StateDir, "bench")
	}
	if err := os.MkdirAll(workspace, 0o700); err != nil {
		return nil, fmt.Errorf("bench: the run workspace %s could not be made: %w", workspace, err)
	}
	logf := opts.Logf
	if logf == nil {
		logf = func(string, ...any) {}
	}
	return &RunService{
		db: db, arts: arts, reg: reg, pkgs: opts.Packages, access: opts.Access,
		traffic: opts.Traffic, workspace: workspace, logf: logf,
		cancels: map[string]context.CancelFunc{}, guards: map[string]*guardrails{}, now: time.Now,
	}, nil
}

// Workspace is the station-owned directory a run's files live under.
func (s *RunService) Workspace() string { return s.workspace }

// RecoverOutcome is what a recovery pass found.
type RecoverOutcome struct {
	// Runs lists the runs that had an attempt recovered.
	Runs []string `json:"runs"`
	// Attempts is how many attempt rows moved to outcome-unknown.
	Attempts int `json:"attempts"`
	// At is when the pass ran.
	At time.Time `json:"at"`
}

// RecoverRuns moves every run that was still going when this process started its in-flight
// attempts into the outcome-unknown state, and interrupts the run rows. It is called by a
// station that rebuilt its services over an existing store: it changes nothing about what
// was already measured, and it does not re-drive anything.
//
// The reason it cannot re-drive: an HTTP request that the station cannot prove did or did
// not reach an endpoint may have been answered, billed, cached, rate limited or counted.
// Sending it again would make the recovered run a different measurement than the one that
// was recorded, which is exactly what a durable run exists to prevent. A rerun is an
// explicit act, and Rerun makes new attempts and keeps the old ones.
func (s *RunService) RecoverRuns(ctx context.Context, actor string) (RecoverOutcome, error) {
	if strings.TrimSpace(actor) == "" {
		actor = "station"
	}
	out := RecoverOutcome{At: s.now()}
	runs, err := listRuns(ctx, s.db, 1000)
	if err != nil {
		return out, err
	}
	for _, run := range runs {
		if run.State != RunStateRunning && run.State != RunStateQueued {
			continue
		}
		changed, err := recoverAttempts(ctx, s.db, actor, run.RunID, out.At)
		if err != nil {
			return out, err
		}
		out.Attempts += changed
		out.Runs = append(out.Runs, run.RunID)
		detail := run.Detail
		if changed > 0 {
			detail.Notes = append(detail.Notes, fmt.Sprintf(
				"%d attempt(s) were open when the station stopped and are recorded as outcome-unknown; they were not re-run",
				changed))
		}
		detail.StoppedBy = "station stopped"
		if err := setRunState(ctx, s.db, actor, run.RunID, RunStateInterrupted,
			"the station stopped while this run was going", detail, out.At); err != nil {
			return out, err
		}
	}
	return out, nil
}

// Cancel asks a run to stop. When this process is driving it, its context is cancelled and
// it ends now; either way the durable flag is set, so a run driven by another process sees
// the request too.
func (s *RunService) Cancel(ctx context.Context, runID, actor string) error {
	if err := requestRunCancel(ctx, s.db, actor, runID); err != nil {
		return err
	}
	s.mu.Lock()
	cancel := s.cancels[runID]
	s.mu.Unlock()
	if guards := s.stopCause(runID); guards != nil {
		// Named here rather than inferred later: an operator stopped this run, and a run
		// that ended with a cancelled context and no cause recorded would leave a reader to
		// guess whether that was an operator or a ceiling.
		guards.markCancelled()
	}
	if cancel != nil {
		cancel()
	}
	return nil
}

// Progress reads a run's progress from its rows, which any reader can do, whether or not
// this process is driving it.
func (s *RunService) Progress(ctx context.Context, runID string) (RunProgress, error) {
	return readProgress(ctx, s.db, runID)
}

// Read runs and attempts.

// ReadRun reads one run row with every attempt and each attempt's record.
func (s *RunService) ReadRun(ctx context.Context, runID string) (RunRow, []AttemptRow, error) {
	run, err := readRun(ctx, s.db, runID)
	if err != nil {
		return run, nil, err
	}
	attempts, err := readAttempts(ctx, s.db, runID)
	if err != nil {
		return run, nil, err
	}
	return run, attempts, nil
}

// ReadEvidence lists the artifacts a run recorded.
func (s *RunService) ReadEvidence(ctx context.Context, runID string) ([]EvidenceRow, error) {
	return readEvidence(ctx, s.db, runID)
}

// ReadCases reads the cases a run generated, in the run's own order, from the case keys
// recorded in its detail document.
func (s *RunService) ReadCases(ctx context.Context, runID string) ([]CaseRow, []Case, error) {
	run, err := readRun(ctx, s.db, runID)
	if err != nil {
		return nil, nil, err
	}
	keys := run.Detail.Documents[documentCaseKeys]
	if keys == "" {
		return nil, nil, nil
	}
	raw, err := s.arts.Bytes(ctx, keys)
	if err != nil {
		return nil, nil, err
	}
	var list []string
	if err := json.Unmarshal(raw, &list); err != nil {
		return nil, nil, fmt.Errorf("bench: run %s holds a case key list this build cannot read: %w", runID, err)
	}
	rows, err := readCases(ctx, s.db, list)
	if err != nil {
		return nil, nil, err
	}
	cases := make([]Case, 0, len(rows))
	for _, row := range rows {
		if row.PromptArtifactKey == "" {
			continue
		}
		body, err := s.arts.Bytes(ctx, row.PromptArtifactKey)
		if err != nil {
			return rows, nil, err
		}
		var c Case
		if err := json.Unmarshal(body, &c); err != nil {
			return rows, nil, fmt.Errorf("bench: the recorded prompt of case %s is not a case: %w", row.Name, err)
		}
		cases = append(cases, c)
	}
	return rows, cases, nil
}

// ReadAttemptRaw reads back one attempt's raw payload bytes as they were kept.
func (s *RunService) ReadAttemptRaw(ctx context.Context, key string) ([]byte, error) {
	return s.arts.Bytes(ctx, key)
}

// The run's own documents, named by the keys of RunDetail.Documents.
const (
	documentPins       = "pins"
	documentRecord     = "record"
	documentReport     = "report"
	documentTraffic    = "traffic"
	documentHardware   = "hardware"
	documentCaseKeys   = "caseKeys"
	documentCaseDigest = "caseDigests"
)

// RunOutcome is what a finished run hands back. Everything in it is also on disk.
type RunOutcome struct {
	RunID     string         `json:"runId"`
	State     string         `json:"state"`
	StoppedBy string         `json:"stoppedBy,omitempty"`
	Detail    RunDetail      `json:"detail"`
	Attempts  int            `json:"attempts"`
	Warnings  []string       `json:"warnings,omitempty"`
	ErrorText string         `json:"errorText,omitempty"`
	Report    *Report        `json:"-"`
	Staged    *StagedPackage `json:"staged,omitempty"`
}

// runState is one run in flight. It is not exported: a caller sees RunSpec in and
// RunOutcome out, and everything in between is recorded rather than remembered.
type runState struct {
	spec        RunSpec
	id          string
	version     *BenchmarkVersion
	stored      StoredVersion
	gen         Generator
	executor    Executor
	execID      extensions.ID
	exportID    extensions.ID
	export      Export
	scorers     []scoredScorer
	cases       []Case
	caseKeys    []string
	secrets     *SecretSet
	guards      *guardrails
	concurrency int
	deadline    time.Time
	startedAt   time.Time

	mu               sync.Mutex
	detail           RunDetail
	records          []CaseRecord
	agenticRecords   map[string]CaseRecord
	agenticPhases    map[string]string
	agenticWindowEnd time.Time
	notes            []OperationNote
	warnings         []string

	requestTimeout time.Duration
	caseTimeout    time.Duration
	traffic        *TrafficObservation
	staged         *StagedPackage
	evalSpec       *EvaluatorSpec
	client         HTTPClient
	// cancel ends the run's own context. A ceiling, a cancellation and the run's own
	// finish all reach it through this field, so there is one way a run stops.
	cancel context.CancelFunc

	// recordCtx is the caller's context, which nothing the run does to itself cancels. Every
	// write a run makes goes through it, so a stopped run still writes down what it did.
	recordCtx context.Context

	// The rerun fields carry the pass this one follows. A rerun adds attempts to the same
	// run, so what the earlier pass recorded has to survive the write of the new account.
	prior           *RunDetail
	priorNotes      []string
	priorReruns     []RerunRecord
	firstNewAttempt int
}

// scoredScorer is one scorer with the id it was looked up under, so an attempt record says
// which ids judged it.
type scoredScorer struct {
	id extensions.ID
	sc Scorer
}

// Run drives one durable run. It returns the outcome and an error together when the run
// ended badly: the outcome is still the record of what happened, which is the point of
// recording it.
func (s *RunService) Run(ctx context.Context, spec RunSpec) (RunOutcome, error) {
	return s.runJob(ctx, spec, s.now(), nil)
}

// Rerun drives a recorded run again, under the same run id, making NEW attempts.
//
// A rerun is an explicit act, and it is the only way an attempt is ever driven twice: a
// recovery pass never calls it. What the run was pinned to is read back out of the pins
// document it recorded, so a rerun measures against the same deployment generation, version,
// package and ceilings rather than against whatever the caller remembers. The request headers
// are the one thing a rerun is handed again, because a run's secrets are deliberately not
// written down anywhere.
func (s *RunService) Rerun(ctx context.Context, runID, actor string, headers map[string]string) (RunOutcome, error) {
	row, err := readRun(ctx, s.db, runID)
	if err != nil {
		return RunOutcome{}, err
	}
	if strings.TrimSpace(actor) == "" {
		return RunOutcome{}, fmt.Errorf("bench: a rerun of run %s names who asked for it", runID)
	}
	s.mu.Lock()
	_, live := s.cancels[runID]
	s.mu.Unlock()
	if live {
		return RunOutcome{}, fmt.Errorf("bench: run %s is being driven by this process, so it cannot be rerun by it as well", runID)
	}
	key := row.Detail.Documents[documentPins]
	if key == "" {
		return RunOutcome{}, fmt.Errorf("bench: run %s recorded no pins document, so a rerun cannot say what it would pin again", runID)
	}
	raw, err := s.arts.Bytes(ctx, key)
	if err != nil {
		return RunOutcome{}, fmt.Errorf("bench: read back the pins of run %s: %w", runID, err)
	}
	var pins Pins
	if err := json.Unmarshal(raw, &pins); err != nil {
		return RunOutcome{}, fmt.Errorf("bench: the pins recorded by run %s are not pins this build can read: %w", runID, err)
	}
	executor, ok := findVersion(pins.ObservedVersions, KindExecutor)
	if !ok {
		return RunOutcome{}, fmt.Errorf("bench: run %s recorded no observed executor version, so a rerun cannot know what drove it", runID)
	}
	spec := RunSpec{
		Actor: actor, RunID: runID, Kind: row.Kind, Pins: pins, Target: row.Detail.Target,
		Executor: executor.ID, Headers: headers, Ceilings: row.Detail.Ceilings.Ceilings,
		Isolation: row.Detail.Isolation, Capabilities: row.Detail.Capabilities,
		WarnConcurrentRequests: row.Detail.WarnConcurrentRequests,
		TrafficWindow:          time.Duration(row.Detail.TrafficWindowNanos),
		Agentic:                row.Detail.AgenticPlan,
	}
	if err := spec.Validate(); err != nil {
		return RunOutcome{}, fmt.Errorf("bench: run %s cannot be rerun as it stands: %w", runID, err)
	}
	return s.runJob(ctx, spec, s.now(), &row.Detail)
}

// runJob drives one pass of a run. A nil prior is a first pass, which writes the run row; a
// non-nil prior is a rerun of the run that detail describes, which keeps that pass's
// documents and notes and adds attempt rows of its own.
func (s *RunService) runJob(ctx context.Context, spec RunSpec, at time.Time, prior *RunDetail) (RunOutcome, error) {
	if err := spec.Validate(); err != nil {
		return RunOutcome{}, err
	}
	// Two contexts. The recording context is the caller's, and nothing the run does to
	// itself reaches it: a run that is cancelled or stopped by a ceiling still writes down
	// everything it measured, which is the whole point of recording it. The driving context
	// is the one the bounds and the cancellation act on, and it only ever drives.
	recording := ctx
	ctx, cancel := context.WithCancel(recording)
	defer cancel()

	rs, err := s.prepare(ctx, spec, at)
	if err != nil {
		return RunOutcome{}, err
	}
	rs.recordCtx = recording
	rs.cancel = cancel
	rs.deadline = at.Add(spec.Ceilings.WallClock)
	rs.startedAt = at
	rs.guards, rs.concurrency = newGuardrails(spec.Ceilings, runnerConcurrency(rs.stored), at, cancel)
	rs.detail = RunDetail{
		AgenticPlan:            rs.spec.Agentic,
		Isolation:              rs.spec.Isolation,
		Capabilities:           rs.spec.Capabilities,
		Target:                 rs.spec.Target,
		Documents:              map[string]string{},
		WarnConcurrentRequests: rs.spec.WarnConcurrentRequests,
		TrafficWindowNanos:     int64(rs.spec.TrafficWindow),
	}
	if prior != nil {
		// A rerun keeps naming what the first pass wrote. Rerunning does not unwrite a
		// record, and every artifact the earlier pass left stays named as this run's
		// evidence; the new pass adds its own documents beside them.
		for name, doc := range prior.Documents {
			rs.detail.Documents[name] = doc
		}
		rs.prior = prior
		rs.priorNotes = append(rs.priorNotes, prior.Notes...)
		rs.priorReruns = append(rs.priorReruns, prior.Reruns...)
	}

	s.bind(rs)
	defer s.release(rs.id)

	if prior == nil {
		if _, err := createRun(rs.recordCtx, s.db, rs.spec.Actor, rs.id, rs.spec.Kind, rs.version.ID(),
			rs.spec.Pins.Deployment, at); err != nil {
			return RunOutcome{}, err
		}
	} else if first, err := nextAttemptNo(recording, s.db, rs.id); err == nil {
		rs.firstNewAttempt = first
	}
	rs.detail.Ceilings = rs.guards.report(rs.concurrency)
	if err := setRunState(rs.recordCtx, s.db, rs.spec.Actor, rs.id, RunStateRunning, "", rs.detail, at); err != nil {
		return RunOutcome{RunID: rs.id}, err
	}

	// The documents that say what this run was pinned to are written before anything is
	// driven. A run that never got a single request answered still says what it was trying
	// to measure.
	if err := rs.recordHardware(ctx, s); err != nil {
		return s.failRun(ctx, rs, "hardware snapshot", err)
	}
	if err := rs.recordPins(ctx, s); err != nil {
		return s.failRun(ctx, rs, "run pins", err)
	}

	if err := rs.observeTraffic(ctx, s); err != nil {
		// A failed probe is recorded and does not stop the run: not knowing whether the
		// endpoint was busy is a condition of the run, not a defect in it.
		rs.warn("the station could not observe the endpoint's other traffic: %v", err)
	}

	if err := rs.stagePackage(ctx, s); err != nil {
		return s.refuseRun(ctx, rs, "package approval", err)
	}
	if err := rs.generateCases(ctx, s); err != nil {
		return s.refuseRun(ctx, rs, "case generation", err)
	}

	rs.drive(ctx, s)

	return s.finishRun(ctx, rs)
}

// bind records the live run so Cancel can reach it.
func (s *RunService) bind(rs *runState) {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.cancels[rs.id] = rs.cancel
	s.guards[rs.id] = rs.guards
}

// release forgets a run that has ended.
func (s *RunService) release(runID string) {
	s.mu.Lock()
	defer s.mu.Unlock()
	delete(s.cancels, runID)
	delete(s.guards, runID)
}

// stopCause names the live run's stop cause, for a cancellation this process received.
func (s *RunService) stopCause(runID string) *guardrails {
	s.mu.Lock()
	defer s.mu.Unlock()
	return s.guards[runID]
}

// finishRun ends a run: it accounts for what was measured, writes the documents that hold
// the rest of the record, and moves the run row to its terminal state.
//
// The state is the job's state, not the model's grade. A run that drove every case it
// planned and got nothing but transport failures finished its work, so it reads as
// succeeded with an aggregate that says nothing was measured; a run that a ceiling or an
// error stopped reads as failed, and a run somebody cancelled reads as cancelled.
func (s *RunService) finishRun(ctx context.Context, rs *runState) (RunOutcome, error) {
	finished := s.now()
	stopped := rs.guards.stoppedBy()

	rs.mu.Lock()
	records := append([]CaseRecord(nil), rs.records...)
	detail := rs.detail
	rs.mu.Unlock()
	warnings, notes := rs.snapshot()

	requests, output, artifacts := rs.guards.counts()
	agg := &detail.Aggregate
	agg.RequestsMade, agg.OutputBytes, agg.ArtifactBytes = requests, output, artifacts
	driven := 0
	for _, rec := range records {
		if rs.spec.Agentic != nil && (rec.Phase == "warmup" || rec.Phase == "drain") {
			continue
		}
		if len(rec.Verdicts) > 0 {
			driven++
		}
	}
	if agg.CasesPlanned < driven {
		// The plan is what the sealed version fixed. If more attempts were driven than that
		// says, the plan is the number that was wrong, and it is corrected rather than used
		// to hide attempts under an unavailable count of zero.
		agg.CasesPlanned = driven
	}
	agg.Unavailable = agg.CasesPlanned - driven
	tallyRun(rs.version.ID(), agg, records, nil)

	detailNotes := append([]string(nil), rs.priorNotes...)
	detailNotes = append(detailNotes, noteDetails(notes)...)
	detailNotes = append(detailNotes, notesOfCeilings(detail.Ceilings.Hits)...)
	if stopped != "" {
		detailNotes = append(detailNotes, "the run stopped because it reached "+stopped)
	}
	detail.Notes = detailNotes
	detail.Reruns = append([]RerunRecord(nil), rs.priorReruns...)
	if rs.prior != nil {
		// The rerun says which attempts it added, so a reader of the run can tell the pass
		// each attempt came from without opening every record.
		last := 0
		for _, rec := range records {
			if rec.AttemptNo > last {
				last = rec.AttemptNo
			}
		}
		detail.Reruns = append(detail.Reruns, RerunRecord{
			At: finished, Actor: rs.spec.Actor, FirstAttempt: rs.firstNewAttempt,
			LastAttempt: last, Attempts: len(records),
		})
	}
	detail.Warnings = warnings
	detail.StoppedBy = stopped
	detail.Ceilings = rs.guards.report(rs.concurrency)
	detail.Traffic = rs.traffic
	if rs.detail.Refusal != nil {
		detail.Refusal = rs.detail.Refusal
	}

	// The full record: every attempt's record in one document, beside the rows that hold
	// the same account. Then the report, which is the same measurements in the shape an
	// export takes. Both are the run's evidence, so both are named as such.
	if key, _, err := rs.writeDocument(ctx, s, ArtifactKindRecord, map[string]any{
		"runId": rs.id, "pins": rs.spec.Pins, "aggregate": *agg, "attempts": records,
	}); err == nil {
		detail.Documents[documentRecord] = key
	} else {
		s.logf("bench: run %s could not record its run document: %v", rs.id, err)
	}

	var report *Report
	if built, err := rs.buildReport(finished); err != nil {
		s.logf("bench: run %s has no report: %v", rs.id, err)
	} else if key, _, err := rs.writeDocument(ctx, s, ArtifactKindReport, built); err != nil {
		s.logf("bench: run %s could not record its report: %v", rs.id, err)
	} else {
		detail.Documents[documentReport] = key
		kept := built
		report = &kept
	}

	state := RunStateSucceeded
	errorText := ""
	switch stopped {
	case "":
	case "cancelled":
		state = RunStateCancelled
		errorText = "the run was cancelled before it drove everything it planned"
	default:
		state = RunStateFailed
		errorText = fmt.Sprintf("the run stopped at its %s ceiling", stopped)
	}

	if n := len(detail.Reruns); rs.prior != nil && n > 0 {
		detail.Reruns[n-1].State = state
	}

	if err := setRunState(rs.recordCtx, s.db, rs.spec.Actor, rs.id, state, errorText, detail, finished); err != nil {
		return RunOutcome{RunID: rs.id, State: state, Detail: detail}, err
	}
	out := RunOutcome{
		RunID: rs.id, State: state, StoppedBy: stopped, Detail: detail,
		Attempts: len(records), Warnings: warnings, ErrorText: errorText,
		Report: report, Staged: rs.staged,
	}
	if state != RunStateSucceeded {
		return out, fmt.Errorf("bench: run %s ended in state %s: %s", rs.id, state, errorText)
	}
	return out, nil
}

// noteDetails is the note text an operator reads back, without the kinds and timestamps the
// attempt records hold in full.
func noteDetails(notes []OperationNote) []string {
	out := make([]string, 0, len(notes))
	for _, n := range notes {
		out = append(out, n.Kind+": "+n.Detail)
	}
	return out
}

// notesOfCeilings states each ceiling the run hit with its bound and the observed value.
func notesOfCeilings(hits []CeilingHit) []string {
	out := make([]string, 0, len(hits))
	for _, h := range hits {
		out = append(out, h.String())
	}
	return out
}

// failRun ends a run that could not be driven at all, recording why. The run row exists by
// then, so the failure is a record rather than an absence.
func (s *RunService) failRun(ctx context.Context, rs *runState, stage string, err error) (RunOutcome, error) {
	return s.endRefused(ctx, rs, stage, err, RunStateFailed)
}

// refuseRun ends a run the station declined to drive: an unapproved package, a generator
// that would not produce the fixed cases. The reason is stored on the run so a refused run
// reads as a refusal rather than as an empty one.
func (s *RunService) refuseRun(ctx context.Context, rs *runState, stage string, err error) (RunOutcome, error) {
	return s.endRefused(ctx, rs, stage, err, RunStateFailed)
}

func (s *RunService) endRefused(ctx context.Context, rs *runState, stage string, cause error, state string) (RunOutcome, error) {
	finished := s.now()
	warnings, notes := rs.snapshot()
	rs.mu.Lock()
	detail := rs.detail
	rs.mu.Unlock()
	detail.Refusal = &RunRefusal{Stage: stage, Reason: cause.Error()}
	detail.StoppedBy = stage
	detail.Warnings = warnings
	detail.Notes = append(noteDetails(notes), stage+": "+cause.Error())
	detail.Ceilings = rs.guards.report(rs.concurrency)
	detail.Aggregate.Unavailable = detail.Aggregate.CasesPlanned - detail.Aggregate.CasesDriven
	if err := setRunState(rs.recordCtx, s.db, rs.spec.Actor, rs.id, state, cause.Error(), detail, finished); err != nil {
		return RunOutcome{RunID: rs.id, State: state, Detail: detail, ErrorText: cause.Error()},
			errors.Join(cause, err)
	}
	return RunOutcome{RunID: rs.id, State: state, StoppedBy: stage, Detail: detail,
		Warnings: warnings, ErrorText: cause.Error()}, cause
}

// runnerConcurrency is the concurrency the sealed benchmark version fixed. The run's own
// ceiling can be lower than that but never higher, which is what newGuardrails resolves.
func runnerConcurrency(stored StoredVersion) int { return stored.Spec.Concurrency }

// prepare resolves everything the run dispatches through, and refuses a run whose pins
// disagree with the sealed version it cites. A version that fixed one seed and a run that
// pins another are two different measurements claiming to be one.
func (s *RunService) prepare(ctx context.Context, spec RunSpec, at time.Time) (*runState, error) {
	stored, err := ReadVersion(ctx, s.db, spec.Pins.BenchmarkVersionID)
	if err != nil {
		return nil, err
	}
	sealed, err := Seal(stored.Spec)
	if err != nil {
		return nil, fmt.Errorf("bench: the recorded version %s does not re-seal: %w", stored.ID, err)
	}
	if sealed.ID() != stored.ID {
		return nil, fmt.Errorf("bench: the recorded version %s re-seals to %s", stored.ID, sealed.ID())
	}
	if err := pinsMatchVersion(spec.Pins, sealed.Spec()); err != nil {
		return nil, err
	}
	if spec.Agentic != nil {
		v := sealed.Spec()
		var params AgenticGeneratorParams
		if err := json.Unmarshal(v.Inputs.Params, &params); err != nil {
			return nil, fmt.Errorf("agentic: sealed generator parameters are invalid: %w", err)
		}
		_, corpusDigest, err := AgenticCorpus()
		if err != nil {
			return nil, err
		}
		if v.Inputs.Revision != AgenticRevision || params.CorpusDigest != corpusDigest || params.Profile.Validate() != nil ||
			params.Profile.WarmupWaves != spec.Agentic.WarmupWaves || params.Profile.MeasureSeconds != int(spec.Agentic.MeasureFor/time.Second) ||
			spec.Agentic.MaxMeasuredStarts != spec.Agentic.Concurrency*params.Profile.MaxSessionsPerC || spec.Agentic.MinSamples != params.Profile.MinSamples ||
			v.Concurrency != spec.Agentic.Concurrency || v.Inputs.CaseCount != spec.Agentic.Concurrency*spec.Agentic.WarmupWaves+spec.Agentic.MaxMeasuredStarts || v.Repetitions != 1 {
			return nil, fmt.Errorf("agentic: level plan differs from the sealed benchmark version")
		}
	}
	rs := &runState{
		spec: spec, id: spec.RunID, version: sealed, stored: stored,
		client:         &http.Client{Timeout: 0},
		agenticRecords: map[string]CaseRecord{}, agenticPhases: map[string]string{},
	}
	if rs.id == "" {
		id, err := newRunID()
		if err != nil {
			return nil, err
		}
		rs.id = id
	}
	if err := rs.resolveProviders(s); err != nil {
		return nil, err
	}
	secrets := NewSecretSet(spec.SecretValues()...)
	if spec.Evaluator != nil {
		// An environment reference names a station variable whose value the child needs.
		// The run has to be able to hand it over and must not be able to write it down,
		// so the value it resolves to goes into the same set the header values go into.
		for _, name := range spec.Evaluator.EnvRefs {
			if v := os.Getenv(name); v != "" {
				secrets.Add(v)
			}
		}
	}
	rs.secrets = secrets
	rs.requestTimeout = sealed.Spec().Timeouts.PerRequest
	rs.caseTimeout = sealed.Spec().Timeouts.PerCase
	rs.traffic = nil
	_ = at
	return rs, nil
}

// pinsMatchVersion checks the pins that restate the sealed version against the version
// itself. A restatement is checked, never trusted: the run record is what an operator
// reads back, and one that disagreed with the version it cited would be two records of
// one measurement. Each refusal names the pin it is about.
func pinsMatchVersion(p Pins, v BenchmarkSpec) error {
	if p.Inputs.Seed != v.Inputs.Seed {
		return fmt.Errorf("bench: the run pins seed %q while benchmark version %s sealed seed %q",
			p.Inputs.Seed, v.Name, v.Inputs.Seed)
	}
	if p.Inputs.CaseCount != v.Inputs.CaseCount {
		return fmt.Errorf("bench: the run pins a case count of %d while benchmark version %s sealed %d",
			p.Inputs.CaseCount, v.Name, v.Inputs.CaseCount)
	}
	if p.Inputs.DatasetDigest != v.Inputs.DatasetDigest {
		return fmt.Errorf("bench: the run pins dataset digest %q while benchmark version %s sealed %q",
			p.Inputs.DatasetDigest, v.Name, v.Inputs.DatasetDigest)
	}
	mine, err := canonicalJSON(p.Inputs.Params)
	if err != nil {
		return err
	}
	theirs, err := canonicalJSON(v.Requests)
	if err != nil {
		return err
	}
	if string(mine) != string(theirs) {
		return fmt.Errorf("bench: the run pins request parameters that are not the ones benchmark version %s sealed", v.Name)
	}
	return nil
}

// resolveProviders looks up every implementation the run dispatches through and records
// the version each one reported, which is the only way those versions can be the run's
// own observation rather than a caller's claim.
func (rs *runState) resolveProviders(s *RunService) error {
	vs := rs.version.Spec()
	gen, err := s.reg.Generators.Lookup(vs.Inputs.Generator)
	if err != nil {
		return fmt.Errorf("bench: the run cites generator %s: %w", vs.Inputs.Generator, err)
	}
	if gen.Revision() != vs.Inputs.Revision {
		return fmt.Errorf("bench: benchmark version %s sealed generator %s revision %q; this registry holds revision %q under that id, so the run would drive cases the version did not fix",
			rs.version.ID(), vs.Inputs.Generator, vs.Inputs.Revision, gen.Revision())
	}
	exec, err := s.reg.Executors.Lookup(rs.spec.Executor)
	if err != nil {
		return fmt.Errorf("bench: the run cites executor %s: %w", rs.spec.Executor, err)
	}
	exp, err := s.reg.Exports.Lookup(DefaultExportID)
	if err != nil {
		return fmt.Errorf("bench: the run needs the report exporter %s: %w", DefaultExportID, err)
	}
	var scorers []scoredScorer
	for _, id := range vs.Scoring.Scorers {
		sc, err := s.reg.Scorers.Lookup(id)
		if err != nil {
			return fmt.Errorf("bench: benchmark version %s cites scorer %s: %w", rs.version.ID(), id, err)
		}
		scorers = append(scorers, scoredScorer{id: id, sc: sc})
	}
	if len(scorers) == 0 {
		return fmt.Errorf("bench: benchmark version %s sealed no scorer, so nothing would judge a single case", rs.version.ID())
	}
	rs.gen, rs.executor, rs.execID, rs.exportID, rs.export, rs.scorers = gen, exec, rs.spec.Executor, DefaultExportID, exp, scorers

	observed := []ProviderVersion{
		providerVersionOf(gen, KindGenerator),
		providerVersionOf(exec, KindExecutor),
		providerVersionOf(exp, KindExport),
	}
	for _, sc := range scorers {
		observed = append(observed, providerVersionOf(sc.sc, KindScorer))
	}
	rs.spec.Pins.ObservedVersions = sortVersions(observed)
	rs.spec.Pins.Inputs.HeaderNames = rs.spec.HeaderNameList()
	return nil
}

// providerVersionOf reads one implementation's own version with the role it fills here.
func providerVersionOf(p extensions.Provider, role Kind) ProviderVersion {
	return ProviderVersion{ID: p.Descriptor().ID, ImplVersion: p.Descriptor().ImplVersion, Role: string(role)}
}

// newRunID issues a run id. The bytes come from the system; an id that repeated across two
// runs would put two runs' attempts in one key space.
func newRunID() (string, error) {
	var raw [12]byte
	if _, err := rand.Read(raw[:]); err != nil {
		return "", fmt.Errorf("bench: issue a run id: %w", err)
	}
	return RunIDPrefix + hex.EncodeToString(raw[:]), nil
}
